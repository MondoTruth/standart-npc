//===== rAthena Script =======================================
//= Card Exchanger
//===== By: MondoTruth ========================================
//===== Description: =========================================
//= Trades cards in bulk for a Card Album, and pulls one card
//= of your choice out of equipped gear.
//=   10 cards -> Old Card Album
//=   30 cards -> Mystical Card Album
//= Pulling one card needs app 1.4.8+ (successremovecards with
//= a card slot, Flux159/rathena#25). The item stays equipped
//= and keeps its other cards, refine and options.
//===== How to customize: ====================================
//= - Card counts: appear in 3 places per option (menu text,
//=   the "< 10"/"< 30" check, and .@need) -- change all 3.
//= - Reward item IDs: the getitem lines (616 Old Card Album,
//=   12246 Mystical Card Album).
//= - Extraction fee: .@fee below (currently free, 0z).
//============================================================

prontera,151,187,4	script	Card Exchanger	660,{

	mes "[Card Exchanger]";
	mes "I trade cards in bulk, no questions asked:";
	mes "- 10 cards -> Old Card Album";
	mes "- 30 cards -> Mystical Card Album";
	mes "- Or I can pull a card back out of your gear.";
	next;

	// Get inventory list using correct syntax
	getinventorylist;
	
	// Count all cards
	.@cards = 0;
	for( .@i = 0; .@i < @inventorylist_count; .@i++ ) {
		if( getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) == 6 ) {
			.@cards += @inventorylist_amount[.@i];
		}
	}

	mes "[Card Exchanger]";
	mes "You're currently carrying ^0000FF" + .@cards + "^000000 card(s).";
	next;

	.@menu = select("Old Card Album (10 cards):Mystical Card Album (30 cards):Extract a card:Cancel");

	switch(.@menu) {
	case 1: // Old Card Album
		if( .@cards < 10 ) {
			mes "[Card Exchanger]";
			mes "You need at least ^FF000010 cards^000000 for that.";
			mes "You only have ^0000FF" + .@cards + "^000000 card(s).";
			close;
		}

		// Remove 10 cards
		.@need = 10;
		getinventorylist;
		
		for( .@i = 0; .@i < @inventorylist_count && .@need > 0; .@i++ ) {
			if( getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) == 6 ) {
				if( @inventorylist_amount[.@i] >= .@need ) {
					delitem @inventorylist_id[.@i], .@need;
					.@need = 0;
				} else {
					delitem @inventorylist_id[.@i], @inventorylist_amount[.@i];
					.@need -= @inventorylist_amount[.@i];
				}
			}
		}

		// Give reward - Old Card Album (ID: 616)
		if( .@need == 0 ) {
			getitem 616, 1; // Old Card Album
			mes "[Card Exchanger]";
			mes "Here is your ^0000FFOld Card Album^000000!";
			mes "Pleasure doing business!";
			close;
		} else {
			mes "[Card Exchanger]";
			mes "Error: Could not remove all cards.";
			close;
		}

	case 2: // Mystical Card Album
		if( .@cards < 30 ) {
			mes "[Card Exchanger]";
			mes "You need at least ^FF000030 cards^000000 for that.";
			mes "You only have ^0000FF" + .@cards + "^000000 card(s).";
			close;
		}

		// Remove 30 cards
		.@need = 30;
		getinventorylist;
		
		for( .@i = 0; .@i < @inventorylist_count && .@need > 0; .@i++ ) {
			if( getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) == 6 ) {
				if( @inventorylist_amount[.@i] >= .@need ) {
					delitem @inventorylist_id[.@i], .@need;
					.@need = 0;
				} else {
					delitem @inventorylist_id[.@i], @inventorylist_amount[.@i];
					.@need -= @inventorylist_amount[.@i];
				}
			}
		}

		// Give reward - Mystical Card Album (ID: 12246)
		if( .@need == 0 ) {
			getitem 12246, 1; // Mystical Card Album
			mes "[Card Exchanger]";
			mes "Here is your ^0000FFMystical Card Album^000000!";
			mes "Pleasure doing business!";
			close;
		} else {
			mes "[Card Exchanger]";
			mes "Error: Could not remove all cards.";
			close;
		}

	case 3: // Extract one card from equipped gear
		setarray .@eqi[0], EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW, EQI_ARMOR, EQI_HAND_R, EQI_HAND_L, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R;
		setarray .@eqi_name$[0], "Headgear (Top)", "Headgear (Mid)", "Headgear (Low)", "Armor", "Right Hand", "Left Hand", "Garment", "Shoes", "Accessory (Left)", "Accessory (Right)";

		.@menu2$ = "";
		for (.@i = 0; .@i < getarraysize(.@eqi); .@i++) {
			.@menu2$ += .@eqi_name$[.@i];
			if (getequipisequiped(.@eqi[.@i])) .@menu2$ += " ("+getequipname(.@eqi[.@i])+")";
			.@menu2$ += ":";
		}
		.@menu2$ += "Never mind";

		.@pick = select(.@menu2$) - 1;
		if (.@pick >= getarraysize(.@eqi)) close; // "Never mind"
		.@eq = .@eqi[.@pick];

		if (!getequipisequiped(.@eq)) {
			mes "[Card Exchanger]";
			mes "You don't have anything equipped there.";
			close;
		}

		// Forged, brewed and pet items keep other data in their card slots.
		.@card0 = getequipcardid(.@eq, 0);
		if (.@card0 == CARD0_FORGE || .@card0 == CARD0_CREATE || .@card0 == CARD0_PET) {
			mes "[Card Exchanger]";
			mes "That one was made by hand. There are no cards in it to pull out.";
			close;
		}

		// The cards in the item's own slots, in slot order.
		.@slots = getiteminfo(getequipid(.@eq), ITEMINFO_SLOT);
		.@n = 0;
		.@menu3$ = "";
		for (.@s = 0; .@s < .@slots && .@s < 4; .@s++) {
			.@cid = getequipcardid(.@eq, .@s);
			if (.@cid > 0 && getiteminfo(.@cid, ITEMINFO_TYPE) == IT_CARD) {
				.@cslot[.@n] = .@s;
				.@ccard[.@n] = .@cid;
				.@n++;
				.@menu3$ += getitemname(.@cid) + ":";
			}
		}
		.@menu3$ += "Never mind";

		if (.@n < 1) {
			mes "[Card Exchanger]";
			mes "No cards in that one.";
			close;
		}

		mes "[Card Exchanger]";
		mes "That has "+.@n+" card(s) in it. Which one should I pull out?";
		mes "The item and the other cards stay as they are.";
		next;
		.@c = select(.@menu3$) - 1;
		if (.@c >= .@n) close; // "Never mind"

		.@fee = 0; // Zeny cost for the extraction, if you want one
		mes "[Card Exchanger]";
		mes "Pull out ^0000FF"+getitemname(.@ccard[.@c])+"^000000 for "+.@fee+"z?";
		next;
		if (select("Do it:Never mind") == 2) close;
		if (Zeny < .@fee) {
			mes "[Card Exchanger]";
			mes "Come back with more money.";
			close;
		}
		if (!checkweight(.@ccard[.@c], 1)) {
			mes "[Card Exchanger]";
			mes "You can't carry the card. Make some room first.";
			close;
		}

		successremovecards .@eq, .@cslot[.@c];
		// The command says nothing back: look at the slot to see whether it worked.
		if (getequipcardid(.@eq, .@cslot[.@c]) == 0) {
			Zeny -= .@fee;
			mes "[Card Exchanger]";
			mes "There you go.";
		} else {
			mes "[Card Exchanger]";
			mes "That didn't work.";
		}
		close;

	case 4: // Cancel
		mes "[Card Exchanger]";
		mes "Come back anytime you have cards to exchange!";
		close;
	}

OnInit:
	if (callfunc("F_SNPC_Off", "cardexchanger")) disablenpc strnpcinfo(0);	// switched off in the settings window
	end;
}
