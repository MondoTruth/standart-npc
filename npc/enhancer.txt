//===== rAthena Script =======================================
//= Dynamic Safe Refiner (+7 to +15 Certificates)
//===== By: Lil Art ===========================================
//===== Description: =========================================
//= Refines an equipped item straight to the chosen level for
//= one refine certificate of that level.
//= Renewal: +7 to +15. Pre-renewal: +7, +8, +9 and +11 only
//= (the other certificates do not exist there).
//===== How to customize: ====================================
//= - Certificate item IDs (what gets consumed per target level):
//=   the .armor_certs/.weapon_certs arrays below.
//= - Refine levels offered: the .@levels arrays (one for Renewal,
//=   one for Pre-renewal). Each level needs a certificate in both
//=   arrays above.
//= - Location/sprite: the first line (map,x,y,dir / sprite).
//============================================================
prontera,164,172,4	script	Safe Refiner	1930,{
    // Divine Pride Item IDs for Armor Certificates (+7 to +15)
    setarray .armor_certs[7],   6234,  // Safe to +7 Armor Certificate
                                6233,  // Safe to +8 Armor Certificate
                                6232,  // Safe to +9 Armor Certificate
                                6994,  // Safe to +10 Armor Certificate
                                6239,  // Safe to +11 Armor Certificate
                                6585,  // Safe to +12 Armor Certificate
                                6876,  // Safe to +13 Armor Certificate
                                6877,  // Safe to +14 Armor Certificate
                                6878;  // Safe to +15 Armor Certificate

    // Divine Pride Item IDs for Weapon Certificates (+7 to +15)
    setarray .weapon_certs[7],  6230,  // Safe to +7 Weapon Certificate
                                6229,  // Safe to +8 Weapon Certificate
                                6228,  // Safe to +9 Weapon Certificate
                                6993,  // Safe to +10 Weapon Certificate
                                6238,  // Safe to +11 Weapon Certificate
                                6584,  // Safe to +12 Weapon Certificate
                                6870,  // Safe to +13 Weapon Certificate
                                6871,  // Safe to +14 Weapon Certificate
                                6872;  // Safe to +15 Weapon Certificate

    // checkre(0): 1 = Renewal, 0 = Pre-renewal
    if (checkre(0))
        setarray .@levels, 7, 8, 9, 10, 11, 12, 13, 14, 15;
    else
        setarray .@levels, 7, 8, 9, 11;
    for (.@i = 0; .@i < getarraysize(.@levels); .@i++)
        .@levels$ += (.@i ? ":" : "") + "+" + .@levels[.@i];

    mes "[Safe Refiner]";
    mes "Select an equipped item and the target refine level (" + .@levels$ + ").";
    next;

    .@menu = select("Top Headgear:Middle Headgear:Lower Headgear:Body Armor:Right Hand (Weapon):Left Hand (Shield):Garment:Shoes:Right Accessory:Left Accessory:Cancel");

    switch(.@menu) {
        case 1:  .@equip_slot = EQI_HEAD_TOP; break;
        case 2:  .@equip_slot = EQI_HEAD_MID; break;
        case 3:  .@equip_slot = EQI_HEAD_LOW; break;
        case 4:  .@equip_slot = EQI_ARMOR; break;
        case 5:  .@equip_slot = EQI_HAND_R; break;
        case 6:  .@equip_slot = EQI_HAND_L; break;
        case 7:  .@equip_slot = EQI_GARMENT; break;
        case 8:  .@equip_slot = EQI_SHOES; break;
        case 9:  .@equip_slot = EQI_ACC_R; break;
        case 10: .@equip_slot = EQI_ACC_L; break;
        case 11: close;
    }

    if (getequipid(.@equip_slot) == -1) {
        mes "[Safe Refiner]";
        mes "You aren't wearing anything in that slot!";
        close;
    }

    .@current_refine = getequiprefinerycnt(.@equip_slot);

    mes "[Safe Refiner]";
    mes "Select your target refinement level:";
    next;

    .@target_level = .@levels[select(.@levels$) - 1];

    if (.@current_refine >= .@target_level) {
        mes "[Safe Refiner]";
        mes "Your item is already +" + .@current_refine + "! Select a level higher than your item's current refine level.";
        close;
    }

    // Determine Certificate ID based on weapon vs. armor slot
    if (.@equip_slot == EQI_HAND_R) {
        .@req_cert = .weapon_certs[.@target_level];
    } else {
        .@req_cert = .armor_certs[.@target_level];
    }

    // Verify certificate presence
    if (countitem(.@req_cert) < 1) {
        mes "[Safe Refiner]";
        mes "You need 1x ^FF0000" + getitemname(.@req_cert) + "^000000 (ID: " + .@req_cert + ") to refine this item to +" + .@target_level + ".";
        close;
    }

    // Consume Certificate & Apply Refinement
    delitem .@req_cert, 1;
    successrefitem .@equip_slot, .@target_level - .@current_refine;

    specialeffect2 154;
    mes "[Safe Refiner]";
    mes "Success! Your equipment is now ^00FF00+" + getequiprefinerycnt(.@equip_slot) + "^000000.";

    close;

OnInit:
    if (callfunc("F_SNPC_Off", "refiner")) disablenpc strnpcinfo(0);	// switched off in the settings window
    end;
}
