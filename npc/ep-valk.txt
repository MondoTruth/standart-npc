//===== rAthena Script =======================================
//= Valkyrie Episode Clear Ticket Redeemer (Verified DP IDs)
//===== Description: =========================================
//= Placed at Prontera (128, 193), this Valkyrie uses exact 
//= Divine Pride item IDs for Episode Clear Tickets (13 - 20).
//= Prompts players for Ep 17 (17.1 vs 17.2) and sets/completes
//= all underlying rAthena quest flags properly.
//============================================================

prontera,128,193,4	script	Valkyrie#EpsClear	4_F_VALKYRIE,{
    mes "[Valkyrie]";
    mes "Greetings, traveler. I possess the power to grant you full completion of historical main story episodes.";
    mes "Present your ^0055FFEpisode Clear Ticket^000000 and I shall fulfill your quest progression.";
    next;

    // Direct Divine Pride Item IDs
    .@ticket_ids[13] = 1000284; // Ep 13
    .@ticket_ids[14] = 1000285; // Ep 14
    .@ticket_ids[15] = 1000286; // Ep 15
    .@ticket_ids[16] = 1000282; // Ep 16
    .@ticket_ids[17] = 1000287; // Ep 17
    .@ticket_ids[18] = 1000288; // Ep 18
    .@ticket_ids[19] = 1000988; // Ep 19
    .@ticket_ids[20] = 1001214; // Ep 20

    // Base/Job EXP Rewards
    .@b_exp[13] = 5000000;   .@j_exp[13] = 2500000;
    .@b_exp[14] = 10000000;  .@j_exp[14] = 5000000;
    .@b_exp[15] = 20000000;  .@j_exp[15] = 10000000;
    .@b_exp[16] = 35000000;  .@j_exp[16] = 17500000;
    .@b_exp[17] = 50000000;  .@j_exp[17] = 25000000;
    .@b_exp[18] = 80000000;  .@j_exp[18] = 40000000;
    .@b_exp[19] = 120000000; .@j_exp[19] = 60000000;
    .@b_exp[20] = 200000000; .@j_exp[20] = 100000000;

    .@menu$ = "";
    .@has_tickets = 0;

    for (.@ep = 13; .@ep <= 20; .@ep++) {
        .@item_id = .@ticket_ids[.@ep];
        if (countitem(.@item_id) > 0) {
            .@menu$ += "Redeem Episode " + .@ep + " Clear Ticket:";
            .@has_tickets = 1;
        } else {
            .@menu$ += "^777777Episode " + .@ep + " Ticket (Missing)^000000:";
        }
    }
    .@menu$ += "Cancel";

    if (!.@has_tickets) {
        mes "[Valkyrie]";
        mes "You do not have any Episode Clear Tickets (Ep 13 - 20) in your inventory.";
        close;
    }

    .@choice = select(.@menu$);
    if (.@choice > 8) close;

    .@selected_ep = .@choice + 12;
    .@selected_ticket = .@ticket_ids[.@selected_ep];

    if (countitem(.@selected_ticket) < 1) {
        mes "[Valkyrie]";
        mes "You do not possess the Episode " + .@selected_ep + " Clear Ticket!";
        close;
    }

    // Special selection prompt for Episode 17 (17.1 Illusion vs 17.2 Sage's Legacy)
    .@sub_ep = 0;
    if (.@selected_ep == 17) {
        mes "[Valkyrie]";
        mes "Episode 17 contains two distinct story lines. Which portion would you like to clear?";
        next;
        .@sub_choice = select("Episode 17.1 (Illusion):Episode 17.2 (Sage's Legacy & Varmundt):Both 17.1 & 17.2:Cancel");
        if (.@sub_choice == 4) close;
        .@sub_ep = .@sub_choice;
    } else {
        mes "[Valkyrie]";
        mes "Are you sure you wish to consume your ^0055FFEpisode " + .@selected_ep + " Clear Ticket^000000?";
        next;
        if (select("Yes, complete the episode!:Cancel") == 2) close;
    }

    if (countitem(.@selected_ticket) < 1) close;

    // Consume Ticket
    delitem .@selected_ticket, 1;

    // Complete Episode Quest Flags (Sets and completes so @checkquest shows 2)
    switch (.@selected_ep) {
        case 13:
            setquest 7000; completequest 7000;
            setquest 7001; completequest 7001;
            break;
        case 14:
            setquest 7200; completequest 7200;
            setquest 7201; completequest 7201;
            break;
        case 15:
            setquest 9000; completequest 9000;
            setquest 9001; completequest 9001;
            break;
        case 16:
            // Banquet for Heroes (16.1) & Terra Gloria (16.2)
            setquest 11200; completequest 11200;
            setquest 11201; completequest 11201;
            setquest 11333; completequest 11333;
            setquest 11354; completequest 11354;
            break;
        case 17:
            if (.@sub_ep == 1 || .@sub_ep == 3) {
                // 17.1 Illusion
                setquest 16000; completequest 16000;
                setquest 16001; completequest 16001;
            }
            if (.@sub_ep == 2 || .@sub_ep == 3) {
                // 17.2 Sage's Legacy / Varmundt
                setquest 16010; completequest 16010;
                setquest 16011; completequest 16011;
            }
            break;
        case 18:
            setquest 18000; completequest 18000;
            setquest 18001; completequest 18001;
            break;
        case 19:
            setquest 19000; completequest 19000;
            setquest 19001; completequest 19001;
            break;
        case 20:
            setquest 20000; completequest 20000;
            setquest 20001; completequest 20001;
            break;
    }

    // Grant EXP
    getexp .@b_exp[.@selected_ep], .@j_exp[.@selected_ep];

    specialeffect2 154;
    mes "[Valkyrie]";
    mes "It is done! Episode " + .@selected_ep + " has been marked as fully cleared, and your rewards have been granted.";
    close;
    end;

OnInit:
    // Switched off in the settings window? Hide.
    if (callfunc("F_SNPC_Off", "epvalk")) disablenpc strnpcinfo(0);
    end;
}
