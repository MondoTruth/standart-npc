//===== rAthena Script =======================================
//= Episode Valkyrie (Episode Clear Tickets)
//===== By: Lil Art, rework MondoTruth ===========================
//===== Description: =========================================
//= Prontera (128,193). Uses an Episode Clear Ticket to skip an
//= episode's main story, as on the official servers: the
//= episode's main quests are marked done, which opens what
//= needs the finished story (instances, dailies, areas). No EXP
//= and no quest rewards. Also sells the tickets for Zeny.
//= Only episodes with a real skip are offered; other tickets are
//= kept for later. Renewal only.
//= Episode 18: ep18_main = 57 (its last step), the main quests
//= that are done at the end are completed, the rest erased. This
//= opens Villa of Deception (wolfvill 79,260) and the dailies.
//===== How to customize: ====================================
//= OnInit: .price[<episode>] is the ticket price in Zeny,
//= .ready is the list of episodes with a skip.
//============================================================

prontera,128,193,4	script	Valkyrie#EpsClear	4_F_VALKYRIE,{
	mes "[Valkyrie]";
	mes "Greetings, traveler. With an ^0055FFEpisode Clear Ticket^000000 I can lift the weight of an episode's story from you.";
	mes "Its main quests count as done, so whatever waits for the finished story opens to you. No rewards come with it.";
	next;
	switch (select("Use a ticket:Buy a ticket:Which episodes?:Cancel")) {
	case 1:
		.@ep = callsub(S_Pick, 0);
		if (!.@ep)
			close;
		if (callsub(S_Done, .@ep)) {
			mes "[Valkyrie]";
			mes "You have already finished Episode " + .@ep + ". Keep your ticket.";
			close;
		}
		if (countitem(.ticket[.@ep]) < 1) {
			mes "[Valkyrie]";
			mes "You have no ^0055FFEpisode " + .@ep + " Clear Ticket^000000.";
			close;
		}
		mes "[Valkyrie]";
		mes "Use your ^0055FFEpisode " + .@ep + " Clear Ticket^000000?";
		mes "The main story of Episode " + .@ep + " will be marked as finished. Quests of it you have open are closed.";
		next;
		if (select("Yes, use it:Cancel") == 2)
			close;
		if (countitem(.ticket[.@ep]) < 1)
			close;
		delitem .ticket[.@ep], 1;
		callsub S_Skip, .@ep;
		specialeffect2 154;
		mes "[Valkyrie]";
		mes "It is done. Episode " + .@ep + " is behind you.";
		close;
	case 2:
		.@ep = callsub(S_Pick, 1);
		if (!.@ep)
			close;
		if (callsub(S_Done, .@ep)) {
			mes "[Valkyrie]";
			mes "You have already finished Episode " + .@ep + ". You do not need its ticket.";
			close;
		}
		mes "[Valkyrie]";
		mes "An ^0055FFEpisode " + .@ep + " Clear Ticket^000000 costs " + callfunc("F_InsertComma", .price[.@ep]) + " Zeny.";
		next;
		if (select("Buy it:Cancel") == 2)
			close;
		if (Zeny < .price[.@ep]) {
			mes "[Valkyrie]";
			mes "You do not have enough Zeny.";
			close;
		}
		if (!checkweight(.ticket[.@ep], 1)) {
			mes "[Valkyrie]";
			mes "You are carrying too much.";
			close;
		}
		Zeny -= .price[.@ep];
		getitem .ticket[.@ep], 1;
		mes "[Valkyrie]";
		mes "Here is your ticket.";
		close;
	case 3:
		mes "[Valkyrie]";
		.@n = getarraysize(.ready);
		for (.@i = 0; .@i < .@n; .@i++)
			mes "Episode " + .ready[.@i] + ": ready.";
		mes "Other episodes will follow. If you hold a ticket for one of them, keep it until then.";
		close;
	}
	close;

// Asks for an episode out of .ready. getarg(0): 1 = with prices.
// Returns the episode, or 0 for Cancel.
S_Pick:
	.@n = getarraysize(.ready);
	for (.@i = 0; .@i < .@n; .@i++) {
		.@ep = .ready[.@i];
		if (getarg(0))
			.@menu$ += "Episode " + .@ep + " (" + callfunc("F_InsertComma", .price[.@ep]) + " z):";
		else
			.@menu$ += "Episode " + .@ep + " (tickets: " + countitem(.ticket[.@ep]) + "):";
	}
	.@menu$ += "Cancel";
	.@c = select(.@menu$) - 1;
	if (.@c >= .@n)
		return 0;
	return .ready[.@c];

// Whether the episode's main story is finished.
S_Done:
	switch (getarg(0)) {
	case 18:
		return (ep18_main >= 57);
	}
	return 0;

// Marks the episode's main story as finished.
S_Skip:
	switch (getarg(0)) {
	case 18:
		// Main quests still open are closed, as the story would have.
		.@n = getarraysize(.q18);
		for (.@i = 0; .@i < .@n; .@i++) {
			if (isbegin_quest(.q18[.@i]) == 1)
				erasequest .q18[.@i];
		}
		// The ones a finished story leaves completed.
		.@n = getarraysize(.q18_done);
		for (.@i = 0; .@i < .@n; .@i++) {
			if (isbegin_quest(.q18_done[.@i]) == 0)
				setquest .q18_done[.@i];
			if (isbegin_quest(.q18_done[.@i]) == 1)
				completequest .q18_done[.@i];
		}
		ep18_main = 57;
		break;
	}
	return;

OnInit:
	// Off on Pre-renewal (checkre(0): 1 = Renewal, 0 = Pre-renewal): no
	// Episode Clear Ticket is in its item db. And when switched off in the
	// settings window.
	if (!checkre(0) || callfunc("F_SNPC_Off", "epvalk")) {
		disablenpc strnpcinfo(0);
		end;
	}
	// Episodes with a real skip, in menu order.
	setarray .ready[0], 18;
	// Episode Clear Tickets (EpisodClear13 - EpisodClear18).
	.ticket[13] = 1000284;
	.ticket[14] = 1000285;
	.ticket[15] = 1000286;
	.ticket[16] = 1000282;
	.ticket[17] = 1000287;
	.ticket[18] = 1000288;
	// Ticket prices in Zeny.
	.price[13] = 1000000;
	.price[14] = 2000000;
	.price[15] = 3000000;
	.price[16] = 4000000;
	.price[17] = 5000000;
	.price[18] = 6000000;
	// Episode 18 (npc/re/quests/quests_18.txt): every main-story quest,
	// ep18_main 1 to 57, and those a finished story leaves completed.
	setarray .q18[0],
		8681, 8682, 8683, 8684, 8685, 8686, 8687,
		11696, 11697, 11698, 11699, 11700, 11701, 11702, 11703,
		11704, 11705, 11706, 11707, 11708, 11709, 11710, 11711, 11712,
		11713, 11714, 11715, 11716, 11717, 11719, 11720, 11721, 11722,
		11723, 11724, 18082,
		16567, 16568, 16569, 16570, 16571, 16572, 16573, 16574, 16575,
		16576, 16577, 16578, 16579, 16580,
		17517, 17518, 17519, 17520, 17521, 17525, 18083, 18084, 18085;
	setarray .q18_done[0], 8681, 11720, 11724, 18085;
	end;
}
