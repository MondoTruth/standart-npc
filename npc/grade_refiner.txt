//===== rAthena Script =======================================
//= Grade Refiner
//===== By: Lil Art ===========================================
//===== Description: =========================================
//= Opens the game's own equipment grading UI and refining UI
//= from one NPC in central Prontera, and refines with refine
//= tickets: the +5 to +19 Weapon/Armor Certificates (weapon
//= levels 1-4, armor level 1) and the [ROS] Refine Guarantee
//= Tickets (+13 to +15, level 5 weapons and level 2 armor),
//= with no chance of failure. A ticket never refines past its
//= own level.
//= Renewal only: grading does not exist on Pre-renewal, and the
//= Safe Refiner covers refining there. The settings window
//= hides it on Pre-renewal, and the NPC hides itself there at
//= start-up.
//===== How to customize: ====================================
//= - Location/sprite: the first line (map,x,y,dir / sprite).
//= - Tickets: the arrays at the top of S_Ticket.
//============================================================

prontera,164,166,4	script	Grade Refiner#prontera	864,{
	mes "[Grade Refiner]";
	mes "Hello! What service would you like to access today?";
	next;

	switch(select("Equipment Grading:Equipment Refinement:Refine with a ticket:Cancel")) {
	case 1:
		close2;
		enchantgradeui;
		end;
	case 2:
		close2;
		refineUI;
		end;
	case 3:
		callsub S_Ticket;
		end;
	case 4:
		mes "[Grade Refiner]";
		mes "Come back whenever you need to process your gear!";
		close;
	}

// Refining with a ticket: rAthena's Ticket Refiner (npc/re/merchants/ticket_refiner.txt),
// which the server ships switched off, with the +10 certificates and the [ROS]
// tickets its "// TODO" left out.
S_Ticket:
	disable_items;
	setarray .@cert_weapon[0],
		6864,	// Guarantee_Weapon_19Up
		6875,	// Guarantee_Weapon_18Up
		6874,	// Guarantee_Weapon_17Up
		6873,	// Guarantee_Weapon_16Up
		6872,	// Guarantee_Weapon_15Up
		6871,	// Guarantee_Weapon_14Up
		6870,	// Guarantee_Weapon_13Up
		6584,	// Guarantee_Weapon_12Up
		6238,	// Guarantee_Weapon_11Up
		6993,	// Guarantee_Weapon_10Up
		6228,	// Guarantee_Weapon_9Up
		6229,	// Guarantee_Weapon_8Up
		6230,	// Guarantee_Weapon_7Up
		6231,	// Guarantee_Weapon_6Up
		6456;	// Guarantee_Weapon_5Up
	setarray .@cert_armor[0],
		6865,	// Guarantee_Armor_19Up
		6881,	// Guarantee_Armor_18Up
		6880,	// Guarantee_Armor_17Up
		6879,	// Guarantee_Armor_16Up
		6878,	// Guarantee_Armor_15Up
		6877,	// Guarantee_Armor_14Up
		6876,	// Guarantee_Armor_13Up
		6585,	// Guarantee_Armor_12Up
		6239,	// Guarantee_Armor_11Up
		6994,	// Guarantee_Armor_10Up
		6232,	// Guarantee_Armor_9Up
		6233,	// Guarantee_Armor_8Up
		6234,	// Guarantee_Armor_7Up
		6235,	// Guarantee_Armor_6Up
		6457;	// Guarantee_Armor_5Up
	setarray .@cert_level[0],19,18,17,16,15,14,13,12,11,10,9,8,7,6,5;
	.@size_cert = getarraysize(.@cert_weapon);
	setarray .@ros_weapon[0],
		1002191,	// Gu_5LvWeapon_15Up_ROS
		1002189,	// Gu_5LvWeapon_14Up_ROS
		1002187;	// Gu_5LvWeapon_13Up_ROS
	setarray .@ros_armor[0],
		1002192,	// Gu_2LvArmor_15Up_ROS
		1002190,	// Gu_2LvArmor_14Up_ROS
		1002188;	// Gu_2LvArmor_13Up_ROS
	setarray .@ros_level[0],15,14,13;
	.@size_ros = getarraysize(.@ros_weapon);

	for (.@i = 0; .@i < .@size_cert; ++.@i)
		if (countitem(.@cert_weapon[.@i]) || countitem(.@cert_armor[.@i]))
			.@check = 1;
	for (.@i = 0; .@i < .@size_ros; ++.@i)
		if (countitem(.@ros_weapon[.@i]) || countitem(.@ros_armor[.@i]))
			.@check = 1;
	if (!.@check) {
		mes "[Grade Refiner]";
		mes "You don't have a ^006400refine ticket^000000.";
		mes "With a Weapon or Armor Certificate, or a [ROS] Refine Ticket, I refine your gear up to the ticket's level with no chance of failure.";
		close;
	}
	mes "[Grade Refiner]";
	mes "I can refine an item up to the ^006400same level as your ticket^000000. There's no chance of breaking it.";
	mes "Which equipment would you like to refine?";
	next;
	setarray .@position$[1],"Head upper","Armor","Left hand","Right hand","Robe","Shoes","Accessory 1","Accessory 2","Head middle","Head lower";
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for (.@i = 1; .@i <= 10; ++.@i)
		.@menu$ += (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : .@position$[.@i] + "- [Empty]") + ":";

	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Grade Refiner]";
		mes "You have to equip the item you want to refine.";
		close;
	}
	if (!getequipisenableref(.@part)) {
		emotion ET_OTL;
		mes "[Grade Refiner]";
		mes "Sorry, this item can't be refined.";
		close;
	}
	.@refineitemid = getequipid(.@part);
	.@refinerycnt = getequiprefinerycnt(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@itemtype = getiteminfo(.@refineitemid, ITEMINFO_TYPE);
	if (.@itemtype == IT_WEAPON && getequipweaponlv(.@part) >= 1 && getequipweaponlv(.@part) <= 4) {
		copyarray .@tickets[0], .@cert_weapon[0], .@size_cert;
		copyarray .@levels[0], .@cert_level[0], .@size_cert;
		.@size = .@size_cert;
		.@type$ = "Weapon Certificate";
	} else if (.@itemtype == IT_WEAPON && getequipweaponlv(.@part) == 5) {
		copyarray .@tickets[0], .@ros_weapon[0], .@size_ros;
		copyarray .@levels[0], .@ros_level[0], .@size_ros;
		.@size = .@size_ros;
		.@type$ = "Level 5 Weapon [ROS] Refine Ticket";
	} else if (.@itemtype == IT_ARMOR && getequiparmorlv(.@part) <= 1) {
		copyarray .@tickets[0], .@cert_armor[0], .@size_cert;
		copyarray .@levels[0], .@cert_level[0], .@size_cert;
		.@size = .@size_cert;
		.@type$ = "Armor Certificate";
	} else if (.@itemtype == IT_ARMOR && getequiparmorlv(.@part) == 2) {
		copyarray .@tickets[0], .@ros_armor[0], .@size_ros;
		copyarray .@levels[0], .@ros_level[0], .@size_ros;
		.@size = .@size_ros;
		.@type$ = "Level 2 Armor [ROS] Refine Ticket";
	} else {
		mes "[Grade Refiner]";
		mes "There's no refine ticket for this kind of equipment.";
		close;
	}
	// Only the tickets of that kind you carry.
	.@menu$ = "";
	.@n = 0;
	for (.@i = 0; .@i < .@size; ++.@i) {
		if (countitem(.@tickets[.@i])) {
			.@have[.@n] = .@tickets[.@i];
			.@have_lv[.@n] = .@levels[.@i];
			.@menu$ += getitemname(.@tickets[.@i]) + ":";
			++.@n;
		}
	}
	if (!.@n) {
		emotion ET_THINK;
		mes "[Grade Refiner]";
		mes "For this one you need a ^006400" + .@type$ + "^000000.";
		close;
	}
	mes "[Grade Refiner]";
	mes "Which ticket would you like to use?";
	next;
	.@select = select(.@menu$) - 1;
	.@ticket_id = .@have[.@select];
	.@ticket_lv = .@have_lv[.@select];
	if (getequiprefinerycnt(.@part) >= .@ticket_lv) {
		emotion ET_PROFUSELY_SWEAT;
		mes "[Grade Refiner]";
		mes "This item is already at +" + getequiprefinerycnt(.@part) + ". That ticket only goes up to +" + .@ticket_lv + ".";
		close;
	}
	mes "[Grade Refiner]";
	mes "I'll refine ^006400" + getequipname(.@part) + "^000000 up to ^006400+" + .@ticket_lv + "^000000 with your ^006400" + getitemname(.@ticket_id) + "^000000.";
	mes "Shall I go ahead?";
	next;
	if (select("Yes, please.:Not now.") == 2) {
		mes "[Grade Refiner]";
		mes "Come back whenever you're ready.";
		close;
	}
	if (countitem(.@ticket_id) < 1) {
		mes "[Grade Refiner]";
		mes "The ticket isn't in your inventory anymore.";
		close;
	}
	// The item must still be the one chosen, as it was.
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
		emotion ET_FRET;
		mes "[Grade Refiner]";
		mes "That's not the item you showed me.";
		close;
	}
	delitem .@ticket_id, 1;
	specialeffect EF_SUI_EXPLOSION;
	successrefitem .@part, .@ticket_lv - getequiprefinerycnt(.@part);
	emotion ET_DELIGHT;
	mes "[Grade Refiner]";
	mes "Done! Your gear is now ^006400+" + .@ticket_lv + "^000000.";
	close;

OnInit:
	// Off on Pre-renewal (checkre(0): 1 = Renewal, 0 = Pre-renewal),
	// or switched off in the settings window
	if (!checkre(0) || callfunc("F_SNPC_Off", "grade"))
		disablenpc strnpcinfo(0);
	end;
}
