//===== rAthena Script =======================================
//= Grade Refiner
//===== By: Lil Art ===========================================
//===== Description: =========================================
//= Opens the game's own equipment grading UI and refining UI
//= from one NPC in central Prontera.
//= Renewal only: grading does not exist on Pre-renewal, and the
//= Safe Refiner covers refining there. The settings window
//= hides it on Pre-renewal, and the NPC hides itself there at
//= start-up.
//===== How to customize: ====================================
//= - Location/sprite: the first line (map,x,y,dir / sprite).
//============================================================

prontera,164,166,4	script	Grade Refiner#prontera	864,{
	mes "[Grade Refiner]";
	mes "Hello! What service would you like to access today?";
	next;

	switch(select("Equipment Grading:Equipment Refinement:Cancel")) {
	case 1:
		close2;
		enchantgradeui;
		end;
	case 2:
		close2;
		refineUI;
		end;
	case 3:
		mes "[Grade Refiner]";
		mes "Come back whenever you need to process your gear!";
		close;
	}

OnInit:
	// Off on Pre-renewal (checkre(0): 1 = Renewal, 0 = Pre-renewal),
	// or switched off in the settings window
	if (!checkre(0) || callfunc("F_SNPC_Off", "grade"))
		disablenpc strnpcinfo(0);
	end;
}
