	}

	// ---- Minstrel / Wanderer ----
	if (.@cat == 9) {
		.@c = select("Metallic Sound:Reverberation:Severe Rainstorm:Cancel");
		if (.@c == 4) close;
		if (.@c == 1) { .@sk = 2413; .@lv = 10; .@name$ = "Metallic Sound"; } // WM_METALICSOUND
		if (.@c == 2) { .@sk = 2414; .@lv = 5; .@name$ = "Reverberation"; } // WM_REVERBERATION
		if (.@c == 3) { .@sk = 2418; .@lv = 5; .@name$ = "Severe Rainstorm"; } // WM_SEVERE_RAINSTORM
	}

	// ---- Sorcerer ----
	if (.@cat == 10) {
		.@c = select("Fire Walk:Electric Walk:Earth Grave:Diamond Dust:Poison Buster:Psychic Wave:Cloud Kill:Varetyr Spear:Cancel");
		if (.@c == 9) close;
		if (.@c == 1) { .@sk = 2443; .@lv = 5; .@name$ = "Fire Walk"; } // SO_FIREWALK
		if (.@c == 2) { .@sk = 2444; .@lv = 5; .@name$ = "Electric Walk"; } // SO_ELECTRICWALK
		if (.@c == 3) { .@sk = 2446; .@lv = 5; .@name$ = "Earth Grave"; } // SO_EARTHGRAVE

	// ---- Genetic ----
	if (.@cat == 11) {
		.@c = select("Thorn Trap:Blood Sucker:Spore Explosion:Wall of Thorns:Crazy Weed:Hell's Plant:Illusion Doping:Cancel");

	}

	// The copied level can't be higher than the Reproduce level.
	.@lv = min(.@lv, .@repro);

	// The engine copies into the Reproduce slot only while the Reproduce
	// status is on, so it is on just for the copy.
	sc_start SC__REPRODUCE, 1000, .@repro, 10000, SCSTART_NOAVOID | SCSTART_NOICON;
	.@ok = plagiarizeskill(.@sk, .@lv);
	sc_end SC__REPRODUCE;

	mes "[Plagiarism Master]";
	if (.@ok) {
		mes "Success! You have learned " + .@name$ + " Lv " + .@lv + ".";
	} else {
		mes "The copy failed for skill ID " + .@sk + ".";
	}
	close;

OnInit:
	if (callfunc("F_SNPC_Off", "plagiarism")) disablenpc strnpcinfo(0);	// switched off in the settings window
	end;
}
