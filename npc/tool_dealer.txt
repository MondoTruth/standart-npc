//===== rAthena Script =======================================
//= Tool Dealer
//===== Description: =========================================
//= Standard fixed-price shop.
//===== How to customize: ====================================
//= - Assortment/prices: the item list after the sprite ID, format
//=   <item ID>:<price>,<item ID>:<price>,... -- price -1 means
//=   "use this item's own Buy price from item_db".
//= - Add an item: append another <ID>:-1 (or <ID>:<price>).
//= - Remove an item: delete its <ID>:price pair (and the comma).
//= - Blacksmith Blessing (6635, Renewal only) is added by the helper
//=   below only where the item exists; its price is set there.
//= - Elemental Converters (12114-12117, 5,000z) are added by the
//=   helper only when Lil Art's cash-shop-extended is not loaded:
//=   with it, its Cash Shop sells them. Master Nokzin sends players
//=   here (143,178) -- change his line too if the shop moves.
//============================================================

prontera,143,178,4	shop	Tool Dealer#custom_tool	497,501:-1,502:-1,503:-1,504:-1,505:-1,506:-1,601:-1,602:-1,1750:-1,1065:-1,645:-1,656:-1,657:-1,611:-1,678:-1

// A shop has no OnInit, so this helper switches it off when "tooldealer"
// is unticked in the settings window, and adds the Renewal-only items.
-	script	SNPC_ToolDealerSwitch	-1,{
OnInit:
	if (callfunc("F_SNPC_Off", "tooldealer")) disablenpc "Tool Dealer#custom_tool";
	// Blacksmith Blessing is not in the Pre-renewal item db; listing it in
	// the shop line there logs "Invalid sell item".
	if (getitemname(6635) != "null") npcshopadditem "Tool Dealer#custom_tool", 6635, 7777;
	// Elemental Converters (Fire, Water, Earth, Wind; both eras), unless
	// cash-shop-extended's Cash Shop sells them. Every NPC file is loaded
	// before any OnInit runs, so its NPC is there by now if it ever will be.
	if (!getnpcid(0, "CashPointRewards"))
		npcshopadditem "Tool Dealer#custom_tool", 12114, 5000, 12115, 5000, 12116, 5000, 12117, 5000;
	end;
}
