//===== rAthena Script =======================================
//= Welcoming
//===== By: Lil Art ============================================
//===== Description: =========================================
//= A one-time starter gift for every new character, in Izlude
//= (189,209): 150,000 Zeny, a Premium Booster Box and two
//= costumes. Shows a quest balloon and a minimap mark to a
//= character who has not taken the gift yet, when it comes
//= within 5 cells.
//= Renewal only: Pre-renewal has none of the three items, and
//= 189,209 is not walkable on its old Izlude.
//===== How to customize: ====================================
//= - Zeny: .zeny in OnInit.
//= - Items: .items in OnInit (item IDs, one of each).
//= - Location/sprite: the script line (1096 = Angeling; 5,5 is
//=   the area that shows the balloon).
//= - Per character: SNPC_WelcomeGift. Make it #SNPC_WelcomeGift
//=   for once per account.
//============================================================

izlude,189,209,6	script	Welcoming#starter	1096,5,5,{
	mes "[Welcoming]";
	if (SNPC_WelcomeGift) {
		mes "Welcome back, adventurer!";
		mes "I hope your journey through Midgard is going smoothly.";
		close;
	}

	mes "Hello and welcome to Izlude!";
	mes "I have a special starter gift to help you begin your journey.";
	next;

	mes "[Welcoming]";
	mes "Would you like to receive your beginner starter pack?";
	next;

	if (select("Yes, please!:No, thank you.") == 2) {
		mes "[Welcoming]";
		mes "No problem! Speak to me whenever you are ready.";
		close;
	}

	for (.@i = 0; .@i < getarraysize(.items); .@i++) {
		if (!checkweight(.items[.@i], 1)) {
			mes "[Welcoming]";
			mes "Your bag is too heavy to take the gift. Make some room and come back!";
			close;
		}
	}
	if (Zeny > MAX_ZENY - .zeny) {
		mes "[Welcoming]";
		mes "Your purse is too full to take the Zeny. Spend some and come back!";
		close;
	}

	SNPC_WelcomeGift = 1;
	Zeny += .zeny;
	for (.@i = 0; .@i < getarraysize(.items); .@i++) getitem .items[.@i], 1;

	// Remove the balloon and the minimap mark at once.
	showevent QTYPE_NONE;
	viewpoint 2, 189, 209, 1, 0xFF0000;

	mes "[Welcoming]";
	mes "Here is ^0055FF" + callfunc("F_InsertComma", .zeny) + " Zeny^000000 and your starter reward items!";
	// The box's item db entry has EquipLevelMax 10: it opens only up to Base Lv 10.
	mes "Open the ^0055FFPremium Booster Box^000000 soon: it can only be opened up to Base Level 10!";
	mes "Good luck on your adventures!";
	close;

OnTouch:
	// Fires when a character walks within 5 cells.
	if (!SNPC_WelcomeGift) {
		showevent QTYPE_QUEST2;	// quest balloon
		viewpoint 1, 189, 209, 1, 0xFF0000;	// red mark on the minimap
	} else {
		showevent QTYPE_NONE;
		viewpoint 2, 189, 209, 1, 0xFF0000;
	}
	end;

OnInit:
	// Off on Pre-renewal (checkre(0): 1 = Renewal, 0 = Pre-renewal),
	// and when switched off in the settings window.
	if (!checkre(0) || callfunc("F_SNPC_Off", "welcome")) disablenpc strnpcinfo(0);
	.zeny = 150000;
	// Premium Booster Box, Costume Angeling Bread Hat, Costume Adventurer's Backpack
	setarray .items, 103048, 400115, 20512;
	end;
}
