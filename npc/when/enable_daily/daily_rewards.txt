//===== rAthena Script =======================================
//= Daily Rewards
//===== By: Lil Art ============================================
//===== Description: =========================================
//= One reward every 24 hours, 30 days in a row. Missing more
//= than 48 hours restarts the streak at day 1. Streak and
//= cooldown are account-wide.
//= Prontera (147,166), sprite 10156 (4_M_KOTLIN).
//===== How to customize: ====================================
//= Rewards: .att_items (item IDs) and .att_amounts in OnInit,
//= index = day 1..30.
//============================================================

prontera,147,166,5	script	Daily Rewards#prontera	10156,{

	.@cooldown_sec = 86400; // 24 Hours in seconds
	.@now = gettimetick(2);

	mes "[Daily Attendance Manager]";
	mes "Welcome! Log in every 24 hours to receive your daily attendance rewards.";
	next;

	switch(select("Claim Daily Reward:View 30-Day Attendance Calendar:Cancel")) {
	case 1:
		if (.@now < (#last_attendance_claim + .@cooldown_sec)) {
			.@time_left = (#last_attendance_claim + .@cooldown_sec) - .@now;
			.@hours = .@time_left / 3600;
			.@minutes = ( .@time_left % 3600 ) / 60;
			
			mes "[Daily Attendance Manager]";
			mes "You have already claimed your daily reward!";
			mes "Next claim available in: ^FF0000" + .@hours + "h " + .@minutes + "m^000000.";
			close;
		}

		// Calculate 30-day streak progression (Resets if inactive > 48h)
		if (#attendance_streak == 0 || .@now > (#last_attendance_claim + (2 * .@cooldown_sec))) {
			#attendance_streak = 1;
		} else {
			#attendance_streak++;
			if (#attendance_streak > 30) #attendance_streak = 1;
		}

		.@day = #attendance_streak;
		.@reward_id = .att_items[.@day];
		.@amount = .att_amounts[.@day];

		#last_attendance_claim = .@now;
		getitem .@reward_id, .@amount;

		mes "[Daily Attendance Manager]";
		mes "Day ^00CCFF" + .@day + "^000000 Claimed!";
		mes "You received ^0055FF" + .@amount + "x " + getitemname(.@reward_id) + "^000000.";
		close;

	case 2:
		// Single-Page 30-Day Calendar List
		mes "^0055FF====================================^000000";
		mes "       ^00CCFFDAILY ATTENDANCE CALENDAR^000000";
		mes "^0055FF====================================^000000";
		mes "Current Streak: ^00CCFFDay " + #attendance_streak + " / 30^000000";
		mes "------------------------------------";
		
		for ( .@i = 1; .@i <= 30; .@i++ ) {
			if (.@i == #attendance_streak) {
				.@status$ = "^00CCFF[ACTIVE]^000000";
			} else if (.@i < #attendance_streak) {
				.@status$ = "^888888[CLAIMED]^000000";
			} else {
				.@status$ = "^FF0000[LOCKED]^000000";
			}
			mes "Day " + (.@i < 10 ? "0" : "") + .@i + ": " + .att_amounts[.@i] + "x " + getitemname(.att_items[.@i]) + " " + .@status$;
		}
		mes "^0055FF====================================^000000";
		close;

	case 3:
		mes "[Daily Attendance Manager]";
		mes "Come back every day to claim your rewards!";
		close;
	}

OnPCLogin:
	.@cooldown_sec = 86400;
	if (gettimetick(2) >= (#last_attendance_claim + .@cooldown_sec)) {
		dispbottom "Your Daily Attendance reward is available! Speak to the Attendance NPC in Prontera.";
	}
	end;

OnInit:
	// 4.8.0 kept the rewards in permanent globals; drop the old copies
	deletearray $att_items;
	deletearray $att_amounts;

	// Day 1..30. IDs checked against Flux159/rathena c3231aa (Renewal)
	setarray .att_items[1],
		100906, // Day 1:  E_All_In_One_Buff
		969,    // Day 2:  Gold
		14545,  // Day 3:  Battle_Manual_X3
		12412,  // Day 4:  HE_Bubble_Gum
		12622,  // Day 5:  Boarding_Halter
		6635,   // Day 6:  Blacksmith_Blessing
		6232,   // Day 7:  Guarantee_Armor_9Up
		100742, // Day 8:  Up_Hero_Weapon_Box
		14545,  // Day 9:  Battle_Manual_X3
		100742, // Day 10: Up_Hero_Weapon_Box
		14613,  // Day 11: RWC_Scroll_2012
		410342, // Day 12: Purified_Pigeon_US
		6238,   // Day 13: Guarantee_Weapon_11Up
		6585,   // Day 14: Guarantee_Armor_12Up
		6635,   // Day 15: Blacksmith_Blessing
		1000337,// Day 16: Blessed_Etel_Dust
		1000325,// Day 17: Etel_Skyblue_Jewel
		1000326,// Day 18: Etel_Topaz
		1000327,// Day 19: Etel_Violet_Jewel
		1000328,// Day 20: Etel_Amber
		6878,   // Day 21: Guarantee_Armor_15Up
		1000337,// Day 22: Blessed_Etel_Dust
		6635,   // Day 23: Blacksmith_Blessing
		101653, // Day 24: Viva_Adul_Hat_Box_11
		490337, // Day 25: Amulet_of_GenesisStone
		102070, // Day 26: Justice_Weapon_Box
		102074, // Day 27: GoodnEvil_Helm_Box
		6872,   // Day 28: Guarantee_Weapon_15Up
		102071, // Day 29: Injustice_Weapon_Box
		22829;  // Day 30: Sealed_Card

	// Amount for each day above
	setarray .att_amounts[1],
		15, 7, 3, 5, 1,
		100, 1, 3, 5, 5,
		50, 1, 1, 1, 150,
		50, 250, 250, 250, 250,
		1, 100, 250, 5, 1,
		20, 20, 1, 20, 1;
	end;
}
