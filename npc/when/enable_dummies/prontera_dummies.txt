//===== rAthena Script =======================================
//= Prontera Training Dummies (Medium & Large)
//===== By: Lil Art ===========================================
//===== Description: =========================================
//= Stationary target dummies for testing damage output.
//===== Status: ================================================
//= First release -- expect this to get refined further.
//============================================================

-	script	PrtDummyManager	-1,{
OnInit:
    // Dummy 2411 is Renewal only; on Pre-renewal spawn nothing (see #22, #25).
    if (getmonsterinfo(2411, MOB_LV) < 0) {
        debugmes "Training Dummies: monster 2411 not in this era's mob db, no dummies spawned.";
        end;
    }
    killmonster "prontera", "PrtDummyManager::OnDummyInfo2";
    killmonster "prontera", "PrtDummyManager::OnDummyInfo3";

    // Medium Dummy & Large Dummy
    monster "prontera", 153, 163, "Medium Dummy", 2411, 1, "PrtDummyManager::OnDummyInfo2";
    monster "prontera", 158, 163, "Large Dummy", 2411, 1, "PrtDummyManager::OnDummyInfo3";
    end;

OnDummyInfo2:
    addtimer 2000, "PrtDummyManager::OnRespawn2";
    end;

OnDummyInfo3:
    addtimer 2000, "PrtDummyManager::OnRespawn3";
    end;

OnRespawn2:
    monster "prontera", 153, 163, "Medium Dummy", 2411, 1, "PrtDummyManager::OnDummyInfo2";
    end;

OnRespawn3:
    monster "prontera", 158, 163, "Large Dummy", 2411, 1, "PrtDummyManager::OnDummyInfo3";
    end;
}
