//===== rAthena Script =======================================
//= Eden Group Gramps Turn-In NPC (Extended Brackets)
//===== By: Lil Art ===========================================
//===== Description: =========================================
//= Co-ordinates level-bracketed hunting bounties and map deployment.
//= Supports 7 level brackets with dynamic bounty tracking.
//===== How to customize: ====================================
//= - Per bracket (7 of them): EXP and Zeny rewards are managed in
//=   config_shared_rewards.txt via the F_SNPC_BountyReward function.
//= - Monster target IDs and kill counts (400) are set in the
//=   callsub OnHandleQuest line AND checked separately in OnTIKill
//=   below, so change both places together.
//= - Level requirements: minimum levels appear twice per bracket --
//=   once in the warp menu, once in OnHandleQuest -- keep both in sync.
//= - Monster counts on the maps themselves: gramps_mobs.txt, not
//=   here.
//= - Warp destinations: the warp lines under "MAP WARP LOGIC".
//= - Progress is tracked per CHARACTER (Gramps_Kill1/2,
//=   Gramps_ActiveBracket -- no # prefix).
//= - Players can take lower-tier bounties if above max level with full EXP!
//= - Ongoing bounties can be abandoned via the dynamic main menu option
//=   with target details explicitly highlighted in the confirmation menu.
//============================================================

prontera,147,172,5	script	Gramps	843,{
    .@base_lv = BaseLevel;

    mes "[Gramps]";
    mes "Hello there, adventurer! The Eden Group keeps me busy coordinating hunting bounties for mercenaries.";
    mes "I have level-bracketed Turn-In bounties available. You can even take lower-tier bounties if you prefer!";
    next;

    // Build dynamic menu based on active bounty state
    if (Gramps_ActiveBracket != 0) {
        .@quest_option$ = "Check / Turn-In Bounty:Abandon Ongoing Bounty";
    } else {
        .@quest_option$ = "Check / Take Bounties";
    }

    .@menu = select(.@quest_option$ + ":Warp to Hunting Map:Cancel");

    // ABANDON BOUNTY LOGIC
    if (Gramps_ActiveBracket != 0 && .@menu == 2) {
        // Resolve active bounty details for confirmation
        switch(Gramps_ActiveBracket) {
            case 1: .@b_name$ = "Lv 70-100 (Clock & Anolian)"; .@m1$ = "Clock"; .@m2$ = "Anolian"; break;
            case 2: .@b_name$ = "Lv 101-125 (Nightmare Terror & Blazer)"; .@m1$ = "Nightmare Terror"; .@m2$ = "Blazer"; break;
            case 3: .@b_name$ = "Lv 126-150 (Horn & Antler Scaraba)"; .@m1$ = "Horn Scaraba"; .@m2$ = "Antler Scaraba"; break;
            case 4: .@b_name$ = "Lv 151-175 (Corrupted Robots)"; .@m1$ = "Corrupted Repair Robot"; .@m2$ = "Corrupted Exploration Rover"; break;
            case 5: .@b_name$ = "Lv 176-200 (Corrupted Mine Mobs)"; .@m1$ = "Corrupted Jewelry Ant"; .@m2$ = "Corrupted Abyssman"; break;
            case 6: .@b_name$ = "Lv 201-230 (Corrupted Niflheim Mobs)"; .@m1$ = "Corrupted Murderer"; .@m2$ = "Corrupted Ghost Cube"; break;
            case 7: .@b_name$ = "Lv 231+ (Deep Niflheim Mobs)"; .@m1$ = "Corrupted Grote"; .@m2$ = "Corrupted Blue Moon Loli Ruri"; break;
        }

        mes "[Gramps]";
        mes "Are you sure you want to give up on your current active bounty?";
        mes "Bounty: ^0055FF" + .@b_name$ + "^000000";
        mes "Progress: " + .@m1$ + " (" + Gramps_Kill1 + "/400), " + .@m2$ + " (" + Gramps_Kill2 + "/400)";
        mes "\n^FF0000All progress on this bounty will be permanently lost!^000000";
        next;
        if (select("No, keep bounty:Yes, abandon bounty") == 2) {
            Gramps_ActiveBracket = 0;
            Gramps_Kill1 = 0;
            Gramps_Kill2 = 0;
            mes "[Gramps]";
            mes "Your active bounty for ^0055FF" + .@b_name$ + "^000000 has been abandoned. You are now free to take on a new contract whenever you're ready.";
            close;
        }
        mes "[Gramps]";
        mes "Good focus! Get back out there and finish the job.";
        close;
    }

    // Shift menu indexing if "Abandon Bounty" option was rendered
    if (Gramps_ActiveBracket != 0 && .@menu > 2) {
        .@menu -= 1;
    }

    // CANCEL OPTION
    if (.@menu == 3) {
        mes "[Gramps]";
        mes "Stay safe out there! Come back when you're ready to hunt.";
        close;
    }

    // MAP WARP LOGIC
    if (.@menu == 2) {
        mes "[Gramps]";
        mes "Where would you like to be deployed?";
        next;
        .@warp_choice = select("Level 70+ Ground (alde_dun04):Level 101+ Ground (mag_dun02):Level 126+ Ground (dic_dun02):Level 151+ Ground (verus03):Level 176+ Ground (ein_dun03):Level 201+ Ground (nif_dun01):Level 231+ Ground (nif_dun02):Cancel");

        switch(.@warp_choice) {
            case 1: if (.@base_lv < 70)  { mes "[Gramps]"; mes "You're too weak for this ground! Minimum required level is 70."; close; } warp "alde_dun04", 103, 162; end;
            case 2: if (.@base_lv < 101) { mes "[Gramps]"; mes "You need to be at least Level 101 for this ground!"; close; } warp "mag_dun02", 46, 49; end;
            case 3: if (.@base_lv < 126) { mes "[Gramps]"; mes "You need to be at least Level 126 for this ground!"; close; } warp "dic_dun02", 100, 100; end;
            case 4: if (.@base_lv < 151) { mes "[Gramps]"; mes "You need to be at least Level 151 for this ground!"; close; } warp "verus03", 100, 100; end;
            case 5: if (.@base_lv < 176) { mes "[Gramps]"; mes "You need to be at least Level 176 for this ground!"; close; } warp "ein_dun03", 100, 100; end;
            case 6: if (.@base_lv < 201) { mes "[Gramps]"; mes "You need to be at least Level 201 for this ground!"; close; } warp "nif_dun01", 100, 100; end;
            case 7: if (.@base_lv < 231) { mes "[Gramps]"; mes "You need to be at least Level 231 for this ground!"; close; } warp "nif_dun02", 100, 100; end;
            case 8: close;
        }
    }

    // TURN-IN BOUNTY LOGIC
    mes "[Gramps]";
    mes "Select a level bracket suitable for your current Base Level:";
    next;

    .@bracket = select("Level 70 - 100:Level 101 - 125:Level 126 - 150:Level 151 - 175:Level 176 - 200:Level 201 - 230:Level 231+:Cancel");

    switch(.@bracket) {
        // Lv 70 - 100 Bracket (Clock & Anolian)
        case 1: callsub OnHandleQuest, 1, 70, 100, 1269, 1206, 400, 400, "Clock", "Anolian"; break;
        // Lv 101 - 125 Bracket (Nightmare Terror & Blazer)
        case 2: callsub OnHandleQuest, 2, 101, 125, 1379, 1367, 400, 400, "Nightmare Terror", "Blazer"; break;
        // Lv 126 - 150 Bracket (Horn Scaraba & Antler Scaraba)
        case 3: callsub OnHandleQuest, 3, 126, 150, 2082, 2083, 400, 400, "Horn Scaraba", "Antler Scaraba"; break;
        // Lv 151 - 175 Bracket (Corrupted Repair Robot & Corrupted Exploration Rover)
        case 4: callsub OnHandleQuest, 4, 151, 175, 3248, 3249, 400, 400, "Corrupted Repair Robot", "Corrupted Exploration Rover"; break;
        // Lv 176 - 200 Bracket (Corrupted Jewelry Ant & Corrupted Abyssman)
        case 5: callsub OnHandleQuest, 5, 176, 200, 20598, 20603, 400, 400, "Corrupted Jewelry Ant", "Corrupted Abyssman"; break;
        // Lv 201 - 230 Bracket (Corrupted Murderer & Corrupted Ghost Cube)
        case 6: callsub OnHandleQuest, 6, 201, 230, 20937, 20938, 400, 400, "Corrupted Murderer", "Corrupted Ghost Cube"; break;
        // Lv 231+ Bracket (Corrupted Grote & Corrupted Blue Moon Loli Ruri)
        case 7: callsub OnHandleQuest, 7, 231, 999, 20941, 20940, 400, 400, "Corrupted Grote", "Corrupted Blue Moon Loli Ruri"; break;
        case 8: close;
    }
    end;

// Subroutine: Bounty Handling Framework
OnHandleQuest:
    .@bracket_id = getarg(0);
    .@min_lv     = getarg(1);
    .@max_lv     = getarg(2);
    .@m1         = getarg(3); .@m2 = getarg(4);
    .@req1       = getarg(5); .@req2 = getarg(6);
    .@mob1_name$ = getarg(7); .@mob2_name$ = getarg(8);

    // Minimum Level Check
    if (BaseLevel < .@min_lv) {
        mes "[Gramps]";
        mes "Your level (" + BaseLevel + ") is below the minimum Level " + .@min_lv + " required for this bounty!";
        close;
    }

    if (Gramps_ActiveBracket != 0 && Gramps_ActiveBracket != .@bracket_id) {
        mes "[Gramps]";
        mes "You already have an active bounty in another level bracket! Finish or abandon it first.";
        close;
    }

    // REWARD TURN-IN CHECK
    if (Gramps_Kill1 >= .@req1 && Gramps_Kill2 >= .@req2) {
        .@base_exp = callfunc("F_SNPC_BountyReward", "gramps", .@bracket_id, 0);
        .@job_exp  = callfunc("F_SNPC_BountyReward", "gramps", .@bracket_id, 1);

        if (.@base_exp <= 0 || .@job_exp <= 0) {
            mes "[Gramps]";
            mes "Reward table error -- please tell the server admin.";
            close;
        }

        Gramps_Kill1 = 0;
        Gramps_Kill2 = 0;
        Gramps_ActiveBracket = 0;

        // Route through safe EXP chunking routine
        callfunc("F_SNPC_GiveEXP", .@base_exp, .@job_exp);
        specialeffect2 154;

        mes "[Gramps]";
        mes "Outstanding job! Here is your reward:";
        mes "^0055FF+" + .@base_exp + " Base EXP^000000";
        mes "^0055FF+" + .@job_exp + " Job EXP^000000";
        mes "\n^777777(EXP rewards are multiplied by server rates)^000000";
        close;
    }

    // ALREADY ACCEPTED CHECK
    if (Gramps_ActiveBracket == .@bracket_id) {
        mes "[Gramps]";
        mes "You still have active bounties for this bracket! Current progress:";
        mes "- " + .@mob1_name$ + ": ^0055FF" + Gramps_Kill1 + "^000000 / " + .@req1;
        mes "- " + .@mob2_name$ + ": ^0055FF" + Gramps_Kill2 + "^000000 / " + .@req2;
        close;
    }

    // ACCEPTING NEW BOUNTY
    Gramps_ActiveBracket = .@bracket_id;
    Gramps_Kill1 = 0;
    Gramps_Kill2 = 0;

    mes "[Gramps]";
    mes "Bounty accepted! Defeat " + .@req1 + "x ^0055FF" + .@mob1_name$ + "^000000 and " + .@req2 + "x ^0055FF" + .@mob2_name$ + "^000000, then return to me for your EXP reward.";
    close;
}

// Clean & Cohesive Spawn Controller for TI Maps
alde_dun04	mapflag	cleanmap
mag_dun02	mapflag	cleanmap
dic_dun02	mapflag	cleanmap
verus03		mapflag	cleanmap
ein_dun03	mapflag	cleanmap
nif_dun01	mapflag	cleanmap
nif_dun02	mapflag	cleanmap

-	script	TI_MobManager	-1,{
OnInit:
OnReloadTISpawns:
    cleanmap "alde_dun04"; cleanmap "mag_dun02"; cleanmap "dic_dun02";
    cleanmap "verus03"; cleanmap "ein_dun03"; cleanmap "nif_dun01"; cleanmap "nif_dun02";
    // Spawns moved to gramps_mobs.txt (plain map spawn lines, governed
    // by dynamic_mobs) so they only exist while a player is on the map.
    end;

// Kill Event Trigger
OnTIKill:
    if (Gramps_ActiveBracket == 0) end;

    .@killed_id = killedrid;

    if (Gramps_ActiveBracket == 1) {
        if (.@killed_id == 1269 && Gramps_Kill1 < 400) { Gramps_Kill1 += 1; dispbottom "[Gramps Bounty] Clock: [" + Gramps_Kill1 + "/400]"; }
        if (.@killed_id == 1206 && Gramps_Kill2 < 400) { Gramps_Kill2 += 1; dispbottom "[Gramps Bounty] Anolian: [" + Gramps_Kill2 + "/400]"; }
    }
    else if (Gramps_ActiveBracket == 2) {
        if (.@killed_id == 1379 && Gramps_Kill1 < 400) { Gramps_Kill1 += 1; dispbottom "[Gramps Bounty] Nightmare Terror: [" + Gramps_Kill1 + "/400]"; }
        if (.@killed_id == 1367 && Gramps_Kill2 < 400) { Gramps_Kill2 += 1; dispbottom "[Gramps Bounty] Blazer: [" + Gramps_Kill2 + "/400]"; }
    }
    else if (Gramps_ActiveBracket == 3) {
        if (.@killed_id == 2082 && Gramps_Kill1 < 400) { Gramps_Kill1 += 1; dispbottom "[Gramps Bounty] Horn Scaraba: [" + Gramps_Kill1 + "/400]"; }
        if (.@killed_id == 2083 && Gramps_Kill2 < 400) { Gramps_Kill2 += 1; dispbottom "[Gramps Bounty] Antler Scaraba: [" + Gramps_Kill2 + "/400]"; }
    }
    else if (Gramps_ActiveBracket == 4) {
        if (.@killed_id == 3248 && Gramps_Kill1 < 400) { Gramps_Kill1 += 1; dispbottom "[Gramps Bounty] Corrupted Repair Robot: [" + Gramps_Kill1 + "/400]"; }
        if (.@killed_id == 3249 && Gramps_Kill2 < 400) { Gramps_Kill2 += 1; dispbottom "[Gramps Bounty] Corrupted Exploration Rover: [" + Gramps_Kill2 + "/400]"; }
    }
    else if (Gramps_ActiveBracket == 5) {
        if (.@killed_id == 20598 && Gramps_Kill1 < 400) { Gramps_Kill1 += 1; dispbottom "[Gramps Bounty] Corrupted Jewelry Ant: [" + Gramps_Kill1 + "/400]"; }
        if (.@killed_id == 20603 && Gramps_Kill2 < 400) { Gramps_Kill2 += 1; dispbottom "[Gramps Bounty] Corrupted Abyssman: [" + Gramps_Kill2 + "/400]"; }
    }
    else if (Gramps_ActiveBracket == 6) {
        if (.@killed_id == 20937 && Gramps_Kill1 < 400) { Gramps_Kill1 += 1; dispbottom "[Gramps Bounty] Corrupted Murderer: [" + Gramps_Kill1 + "/400]"; }
        if (.@killed_id == 20938 && Gramps_Kill2 < 400) { Gramps_Kill2 += 1; dispbottom "[Gramps Bounty] Corrupted Ghost Cube: [" + Gramps_Kill2 + "/400]"; }
    }
    else if (Gramps_ActiveBracket == 7) {
        if (.@killed_id == 20941 && Gramps_Kill1 < 400) { Gramps_Kill1 += 1; dispbottom "[Gramps Bounty] Corrupted Grote: [" + Gramps_Kill1 + "/400]"; }
        if (.@killed_id == 20940 && Gramps_Kill2 < 400) { Gramps_Kill2 += 1; dispbottom "[Gramps Bounty] Corrupted Blue Moon Loli Ruri: [" + Gramps_Kill2 + "/400]"; }
    }
    end;
}
