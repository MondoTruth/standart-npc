//===== rAthena Script =======================================
//= Kafra Employee
//===== Description: =========================================
//= Basic Kafra service: save point + storage + identify. Teleport
//= omitted on purpose -- this pack's Warper NPC already covers travel.
//===== How to customize: ====================================
//= - Save point location: the map/x/y in the savepoint line.
//= - NPC location/sprite: the first line (map,x,y,dir / sprite).
//= - Identify cost: currently free -- add a Zeny/item check at
//=   the top of case 3, same pattern as case 1/2, if you want one.
//============================================================

prontera,160,187,4	script	Kafra Employee	4_F_KAFRA1,{
	cutin "kafra_01",2;
	mes "[Kafra Employee]";
	mes "Welcome to Kafra Corp.";
	mes "How may I serve you today?";
	next;

	switch(select("Save:Storage:Identify:Cancel")) {
	case 1:
		savepoint "prontera",156,191;
		mes "[Kafra Employee]";
		mes "Your save point has been set here in Prontera.";
		break;

	case 2:
		mes "[Kafra Employee]";
		mes "Here is your storage.";
		close2;
		cutin "",255;
		openstorage;
		end;

	case 3:
    identifyall;
    mes "[Kafra Employee]";
    mes "All your items have been identified.";
    break;

	case 4:
		mes "[Kafra Employee]";
		mes "Thank you for using Kafra Corp.!";
		break;
	}
	close2;
	cutin "",255;
	end;
}
