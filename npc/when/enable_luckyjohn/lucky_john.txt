//===== rAthena Script =======================================
//= Lucky John
//===== By: IceGlaive ========================================
//= Description:
//= Per-character daily Mercenary Guild area-purge contracts.
//= Assigns 500 monster kills in a level-appropriate zone.
//===== How to customize: ====================================
//= - Rewards: Base EXP, Job EXP, and Zeny rewards are managed in
//=   config_shared_rewards.txt via the F_SNPC_BountyReward function.
//= - Level brackets and map pools: adjust OnAssignRandomMap subroutine.
//= - Kill requirements: change the 500 threshold across OnNPCKillEvent
//=   and main script logic together.
//============================================================

prontera,138,172,5	script	Lucky John	733,{
    .@base_lv = BaseLevel;
    .@today = gettime(7)*10000 + gettime(6)*100 + gettime(5);

    // Active Bounty Check (Per Character)
    if (LuckyJohn_ActiveBracket != 0) {
        .@map_display_name$ = callsub(OnGetMapInfo, LuckyJohn_TargetMap$);

        mes "[Lucky John]";
        mes "Contract Progress Check:";
        mes "Target Sector: ^0055FF" + .@map_display_name$ + "^000000";
        mes "Map Code: ^0055FF" + LuckyJohn_TargetMap$ + "^000000";
        mes "Kills Completed: ^0055FF" + LuckyJohn_Kill1 + "^000000 / 500";
        next;

        if (LuckyJohn_Kill1 >= 500) {
            // Retrieve shared rewards via F_SNPC_BountyReward
            .@reward_base = callfunc("F_SNPC_BountyReward", "lj", LuckyJohn_ActiveBracket, 0);
            .@reward_job  = callfunc("F_SNPC_BountyReward", "lj", LuckyJohn_ActiveBracket, 1);
            .@reward_zeny = callfunc("F_SNPC_BountyReward", "lj", LuckyJohn_ActiveBracket, 2);

            if (.@reward_base <= 0 || .@reward_job <= 0) {
                mes "[Lucky John]";
                mes "Reward table error -- please report this to an Administrator.";
                close;
            }

            // Route through safe EXP chunking routine
            callfunc("F_SNPC_GiveEXP", .@reward_base, .@reward_job);
            Zeny += .@reward_zeny;
            specialeffect2 154;

            .@was_previous_day = (LuckyJohn_AcceptDay < .@today);
            if (!.@was_previous_day)
                LuckyJohn_LastDone = .@today;

            LuckyJohn_ActiveBracket = 0;
            LuckyJohn_TargetMap$ = "";
            LuckyJohn_Kill1 = 0;
            LuckyJohn_AcceptDay = 0;

            mes "[Lucky John]";
            mes "Phenomenal work! Zone cleared successfully.";
            mes "Here is your contract payout:";
            mes "^0055FF+" + .@reward_base + " Base EXP^000000";
            mes "^0055FF+" + .@reward_job + " Job EXP^000000";
            mes "^0055FF+" + .@reward_zeny + " Zeny^000000";
            mes "\n^777777(EXP rewards are multiplied by server rates)^000000";
            next;

            if (.@was_previous_day) {
                mes "[Lucky John]";
                mes "That settles your pending contract--you are free to take on today's contract immediately!";
            } else {
                mes "[Lucky John]";
                mes "All contracts completed for today. See you tomorrow!";
            }
            close;
        }

        .@choice = select("Continue Purge Mission:Abandon Contract:Cancel");
        if (.@choice == 2) {
            mes "[Lucky John]";
            mes "Abandon contract? Your progress (^0055FF" + LuckyJohn_Kill1 + "/500^000000 kills) in ^0055FF" + .@map_display_name$ + "^000000 will be reset!";
            next;
            if (select("No, remain on mission:Yes, abandon contract") == 2) {
                LuckyJohn_ActiveBracket = 0;
                LuckyJohn_TargetMap$ = "";
                LuckyJohn_Kill1 = 0;
                LuckyJohn_AcceptDay = 0;
                mes "[Lucky John]";
                mes "Contract terminated. Speak with me when you're ready for a new location.";
                close;
            }
        }
        close;
    }

    mes "[Lucky John]";
    mes "Greetings, adventurer!";
    mes "Due to widespread reports of monster invasions across the realm, local authorities are overwhelmed.";
    mes "The Mercenary Guild has been contracted to handle large-scale area purges to restore order!";
    next;
    mes "[Lucky John]";
    mes "Instead of hunting down specific targets, your objective is to purge any 500 monsters in your assigned zone to thin their swelling ranks.";
    next;

    // Daily Completion Check
    if (LuckyJohn_LastDone == .@today) {
        mes "[Lucky John]";
        mes "You have already completed today's clearance contract on this character.";
        mes "Come back tomorrow for a new target sector!";
        close;
    }

    if (.@base_lv < 30) {
        mes "[Lucky John]";
        mes "You're a bit green for field purge work! Train up to Level 30 and come back.";
        close;
    }

    mes "[Lucky John]";
    mes "Are you ready to accept today's map clearance contract?";
    next;

    if (select("Accept Contract:Not right now") == 2) {
        mes "[Lucky John]";
        mes "No problem. Drop by whenever you are ready!";
        close;
    }

    callsub OnAssignRandomMap, .@base_lv;
    .@map_display_name$ = callsub(OnGetMapInfo, LuckyJohn_TargetMap$);

    LuckyJohn_Kill1 = 0;
    LuckyJohn_AcceptDay = .@today;

    mes "[Lucky John]";
    mes "Your assignment is locked in!";
    mes "Objective: Defeat ^0055FF500 Monsters^000000 in the target area.";
    mes "Target Location: ^0055FF" + .@map_display_name$ + "^000000";
    mes "Map ID: ^0055FF" + LuckyJohn_TargetMap$ + "^000000";
    mes "\nAny regular monster killed inside this map counts toward your total!";
    close;

// Subroutine: Random Map Assigner
OnAssignRandomMap:
    .@lv = getarg(0);

    if (.@lv >= 30 && .@lv <= 55) {
        LuckyJohn_ActiveBracket = 1;
        .@r = rand(1, 4);
        if (.@r == 1) LuckyJohn_TargetMap$ = "orcsdun01";
        if (.@r == 2) LuckyJohn_TargetMap$ = "pay_dun01";
        if (.@r == 3) LuckyJohn_TargetMap$ = "xmas_dun01";
        if (.@r == 4) LuckyJohn_TargetMap$ = "xmas_dun02";
    } else if (.@lv >= 56 && .@lv <= 75) {
        LuckyJohn_ActiveBracket = 2;
        .@r = rand(1, 7);
        if (.@r == 1) LuckyJohn_TargetMap$ = "pay_dun03";
        if (.@r == 2) LuckyJohn_TargetMap$ = "cmd_fild03";
        if (.@r == 3) LuckyJohn_TargetMap$ = "mosk_fild02";
        if (.@r == 4) LuckyJohn_TargetMap$ = "gon_fild01";
        if (.@r == 5) LuckyJohn_TargetMap$ = "bra_fild01";
        if (.@r == 6) LuckyJohn_TargetMap$ = "mjolnir_04";
        if (.@r == 7) LuckyJohn_TargetMap$ = "um_fild01";
    } else if (.@lv >= 76 && .@lv <= 100) {
        LuckyJohn_ActiveBracket = 3;
        .@r = rand(1, 7);
        if (.@r == 1) LuckyJohn_TargetMap$ = "ice_dun01";
        if (.@r == 2) LuckyJohn_TargetMap$ = "gl_church";
        if (.@r == 3) LuckyJohn_TargetMap$ = "yuno_fild08";
        if (.@r == 4) LuckyJohn_TargetMap$ = "ra_fild12";
        if (.@r == 5) LuckyJohn_TargetMap$ = "mosk_dun02";

        if (.@r == 7) LuckyJohn_TargetMap$ = "gef_fild08";
    } else if (.@lv >= 101 && .@lv <= 125) {
        LuckyJohn_ActiveBracket = 4;
        .@r = rand(1, 4);
        if (.@r == 1) LuckyJohn_TargetMap$ = "ra_fild01";
        if (.@r == 2) LuckyJohn_TargetMap$ = "beach_dun2";
        if (.@r == 3) LuckyJohn_TargetMap$ = "gl_kaval01";
        if (.@r == 4) LuckyJohn_TargetMap$ = "ma_fild02";
    } else if (.@lv >= 126 && .@lv <= 150) {
        LuckyJohn_ActiveBracket = 5;
        .@r = rand(1, 4);
        if (.@r == 1) LuckyJohn_TargetMap$ = "ecl_fild01";
        if (.@r == 2) LuckyJohn_TargetMap$ = "dic_fild01";
        if (.@r == 3) LuckyJohn_TargetMap$ = "moc_fild22";
        if (.@r == 4) LuckyJohn_TargetMap$ = "gl_cas02_";
    } else if (.@lv >= 151 && .@lv <= 175) {
        LuckyJohn_ActiveBracket = 6;
        .@r = rand(1, 4);
        if (.@r == 1) LuckyJohn_TargetMap$ = "ra_fild11";
        if (.@r == 2) LuckyJohn_TargetMap$ = "ba_lost";
        if (.@r == 3) LuckyJohn_TargetMap$ = "rockmi1";
        if (.@r == 4) LuckyJohn_TargetMap$ = "verus02";
    } else if (.@lv >= 176 && .@lv <= 200) {
        LuckyJohn_ActiveBracket = 7;
        .@r = rand(1, 9);
        if (.@r == 1) LuckyJohn_TargetMap$ = "gw_fild01";
        if (.@r == 2) LuckyJohn_TargetMap$ = "gw_fild02";
        if (.@r == 3) LuckyJohn_TargetMap$ = "ba_2whs01";
        if (.@r == 4) LuckyJohn_TargetMap$ = "ba_2whs02";
        if (.@r == 5) LuckyJohn_TargetMap$ = "gl_cas01_";
        if (.@r == 6) LuckyJohn_TargetMap$ = "mag_dun03";
        if (.@r == 7) LuckyJohn_TargetMap$ = "sp_rudus3";
        if (.@r == 8) LuckyJohn_TargetMap$ = "oz_dun01";
        if (.@r == 9) LuckyJohn_TargetMap$ = "oz_dun02";
    } else if (.@lv >= 201 && .@lv <= 230) {
        LuckyJohn_ActiveBracket = 8;
        LuckyJohn_TargetMap$ = "sp_rudus4";
    } else {
        LuckyJohn_ActiveBracket = 9;
        .@r = rand(1, 2);
        if (.@r == 1) LuckyJohn_TargetMap$ = "amicitia1";
        if (.@r == 2) LuckyJohn_TargetMap$ = "amicitia2";
    }
    return;

OnGetMapInfo:
    .@map$ = getarg(0);
    .@name$ = "";
    
    if (.@map$ == "orcsdun01")       .@name$ = "Orc Dungeon Level 1";
    else if (.@map$ == "pay_dun01")  .@name$ = "Payon Cave Level 2";
    else if (.@map$ == "xmas_dun01") .@name$ = "Toy Factory Warehouse";
    else if (.@map$ == "xmas_dun02") .@name$ = "Toy Factory Level 2";

    else if (.@map$ == "pay_dun03")  .@name$ = "Payon Cave Level 4";
    else if (.@map$ == "cmd_fild03") .@name$ = "Zenhai Marsh";

    else if (.@map$ == "mjolnir_04") .@name$ = "Mt. Mjolnir 04";

    else if (.@map$ == "ra_fild12")  .@name$ = "The Plain of Ida";
    else if (.@map$ == "mosk_dun02") .@name$ = "Temny Forest";
    else if (.@map$ == "gef_fild06") .@name$ = "Geffen Field 06";

    else if (.@map$ == "ra_fild01")  .@name$ = "Audhumla Grassland";

    else if (.@map$ == "gl_kaval01") .@name$ = "Glast Heim Chivalry 1";

    else if (.@map$ == "ecl_fild01") .@name$ = "Blooming Flower Land";

    else if (.@map$ == "rockmi1")    .@name$ = "Rock Ridge Mine";
    else if (.@map$ == "verus02")    .@name$ = "Verus Research Building - WISH";

    else if (.@map$ == "sp_rudus3")  .@name$ = "Rudus 3F (Waste Disposal Plant)";

    else if (.@map$ == "oz_dun02")   .@name$ = "Oz Labyrinth 2F";

    else                             .@name$ = .@map$;
    
    return .@name$;
}

// Global Monster Kill Event Listener
-	script	LuckyJohn_KillManager	-1,{
OnNPCKillEvent:
    if (LuckyJohn_ActiveBracket == 0) end;

    if (strcharinfo(3) == LuckyJohn_TargetMap$ && LuckyJohn_Kill1 < 500) {
        LuckyJohn_Kill1 += 1;
        dispbottom "[Lucky John] Purge Progress: [" + LuckyJohn_Kill1 + "/500]";
    }
    end;
}
