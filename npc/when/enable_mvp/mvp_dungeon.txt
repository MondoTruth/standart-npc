//===== rAthena Script =======================================
//= Cheffenia MVP Dungeon Entrance & In-Dungeon Timer Manager
//===== By: Lil Art ===========================================
//===== Description: =========================================
//= Entrance: prontera (140,180). Exit target: prontera (140,178).
//= Pass expiry is a real absolute timestamp (character-bound,
//= CHEFFENIA_EXPIRE), so it keeps ticking in real time whether
//= the player is inside, outside, offline, or dead -- nothing in
//= this script resets it on its own; only running out for real,
//= or the testing menu option below, clears it.
//===== How to customize: ====================================
//= - Duration/price tiers: the two switch(select(...)) blocks in
//=   the Gatekeeper (Cash Points and Zeny). Keep the tier order
//=   the same in both if you want the same duration to line up
//=   with the same case number.
//= - Zeny/Cash Point exchange rate: the "150 CP = 1,000,000z"
//=   ratio baked into the case values and .@zeny_cost formula.
//= - "Reset My Pass" (case 2 under the active-pass menu) clears
//=   CHEFFENIA_EXPIRE for free -- meant for testing only. Remove
//=   this option before giving players real money-cost access,
//=   or anyone can reset their own paid timer for free.
//============================================================

prontera,140,180,4	script	Cheffenia Gatekeeper	10068,{
    .@now = gettimetick(2);

    mes "[Cheffenia Gatekeeper]";
    mes "Welcome to the ^FF0000Cheffenia MVP Dungeon^000000!";
    mes " ";

    // Active Pass Check (Character-Bound)
    if (CHEFFENIA_EXPIRE > .@now) {
        .@remaining_sec = CHEFFENIA_EXPIRE - .@now;
        .@remaining_min = .@remaining_sec / 60;
        mes "Your character's access pass is ^00FF00Active^000000.";
        mes "Time Remaining: ^0000FF" + (.@remaining_min > 0 ? .@remaining_min + " minutes" : .@remaining_sec + " seconds") + "^000000.";
        next;
        switch(select("Enter Dungeon:Reset My Pass (testing only):Cancel")) {
            case 1:
                warp "bossnia_01", 100, 150;
                end;
            case 2:
                CHEFFENIA_EXPIRE = 0;
                mes "[Cheffenia Gatekeeper]";
                mes "Your access pass has been reset for testing.";
                close;
            case 3:
                close;
        }
    }

    mes "Select payment method:";
    next;
    .@currency = select("Cash Points:Zeny:Cancel");
    if (.@currency == 3) close;

    mes "[Cheffenia Gatekeeper]";
    mes "Select access duration:";
    next;

    // Tier Selection Menu with Dynamic Pricing
    if (.@currency == 1) { // Cash Points
        switch(select("60 Minutes (5,000 CP):30 Minutes (3,000 CP):15 Minutes (1,600 CP):5 Minutes (600 CP):60 Seconds (150 CP):Cancel")) {
            case 1: .@cp_cost = 5000; .@duration = 3600; break;
            case 2: .@cp_cost = 3000; .@duration = 1800; break;
            case 3: .@cp_cost = 1600; .@duration = 900;  break;
            case 4: .@cp_cost = 600;  .@duration = 300;  break;
            case 5: .@cp_cost = 150;  .@duration = 60;   break;
            case 6: close;
        }

        if (#CASHPOINTS < .@cp_cost) {
            mes "[Cheffenia Gatekeeper]";
            mes "You do not have enough Cash Points!";
            mes "Required: " + .@cp_cost + " CP | Current: " + #CASHPOINTS + " CP";
            close;
        }
        #CASHPOINTS -= .@cp_cost;

    } else if (.@currency == 2) { // Zeny (150 CP = 1,000,000z)
        switch(select("60 Minutes (33,333,333z):30 Minutes (20,000,000z):15 Minutes (10,666,666z):5 Minutes (4,000,000z):60 Seconds (1,000,000z):Cancel")) {
            case 1: .@cp_cost = 5000; .@duration = 3600; break;
            case 2: .@cp_cost = 3000; .@duration = 1800; break;
            case 3: .@cp_cost = 1600; .@duration = 900;  break;
            case 4: .@cp_cost = 600;  .@duration = 300;  break;
            case 5: .@cp_cost = 150;  .@duration = 60;   break;
            case 6: close;
        }

        .@zeny_cost = .@cp_cost * 1000000 / 150;
        if (Zeny < .@zeny_cost) {
            mes "[Cheffenia Gatekeeper]";
            mes "You do not have enough Zeny!";
            mes "Required: " + .@zeny_cost + "z | Current: " + Zeny + "z";
            close;
        }
        Zeny -= .@zeny_cost;
    }

    // Set character expiration timestamp
    CHEFFENIA_EXPIRE = .@now + .@duration;

    specialeffect2 568;
    dispbottom "[Cheffenia] Pass activated! Real-time countdown started.";

    warp "bossnia_01", 100, 150;
    close;
}

// In-Dungeon Timer NPC Manager
bossnia_01,0,0,0	script	Cheffenia Time Tracker	-1,{

OnPCLoadMapEvent:
    if (strcharinfo(3) != "bossnia_01") end;

    .@now = gettimetick(2);

    // If pass is already expired on entry/relog, trigger dialogue immediately and reset pass
    if (CHEFFENIA_EXPIRE <= .@now) {
        CHEFFENIA_EXPIRE = 0;
        doevent "Cheffenia Time Tracker::OnTimeUp";
        end;
    }

    // Schedule the exact remaining milliseconds until expiration
    .@time_left_ms = (CHEFFENIA_EXPIRE - .@now) * 1000;
    deltimer "Cheffenia Time Tracker::OnTimeUp";
    addtimer .@time_left_ms, "Cheffenia Time Tracker::OnTimeUp";

    .@left = CHEFFENIA_EXPIRE - .@now;
    dispbottom "[Cheffenia] Time remaining: " + (.@left / 60) + "m " + (.@left % 60) + "s.";
    end;

OnTimeUp:
    if (strcharinfo(3) == "bossnia_01") {
        deltimer "Cheffenia Time Tracker::OnTimeUp";
        CHEFFENIA_EXPIRE = 0; // Clear pass variable
        mes "[Cheffenia System]";
        mes "^FF0000TIMES UP!^000000";
        mes "Your access pass has expired.";
        next;
        warp "prontera", 140, 178;
        close;
    }
    end;
}
