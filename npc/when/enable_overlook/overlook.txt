//===== rAthena Script =======================================
//= Overlook Water Dungeon Quests & Card Choice Reward
//===== By: Lil Art ===========================================
//===== How to customize: ====================================
//= - Per floor (4 of them): the callsub OnHandleOWDQuest line sets
//=   monster IDs, kill counts, and EXP reward all at once -- but
//=   the kill count and monster IDs are ALSO checked separately in
//=   OnOWDKill below, so change both places together.
//= - Base Level range to enter (30-85): checked once near "WARP TO
//=   DUNGEON", separate from the per-floor menu labels.
//= - Final card reward choices: the getitem-by-ID switch under
//=   "CLAIM FINAL CARD REWARD".
//= - Monster counts on the maps themselves: overlook_mobs.txt, not
//=   here.
//= - Progress is tracked per CHARACTER (OWD_Kill1/2, OWD_ActiveQuest,
//=   OWD_Floor1-4_Done, OWD_ClearedAll -- no # prefix). Switching
//=   this back to account-wide would mean adding # back to all of
//=   them, everywhere they appear in this file. Same convention as
//=   gramps.txt.
//= - A player with an active bounty can still warp in to finish it
//=   even past Base Level 85 (OWD_ActiveQuest != 0 bypasses the
//=   entry level check) -- the level range only blocks entering
//=   fresh, not finishing an already-active bounty.
//============================================================

// --- Prontera Entrance & Quest Manager (Fisherman Sprite: 100) ---
prontera,147,169,4	script	Overlook Fisherman#owd	100,{
    .@base_lv = BaseLevel;

    mes "[Overlook Fisherman]";
    mes "Ahoy, adventurer! I manage access and bounties for the ^0055FFOverlook Water Dungeon^000000.";
    mes "It's a specialized training ground for adventurers between ^0055FFBase Level 30 and 85^000000.";
    next;

    .@menu = select("Enter Overlook Water Dungeon:Bounty Quests (Accept / Turn-In):Claim Final Card Reward:Information:Cancel");

    if (.@menu == 5) close;

    if (.@menu == 4) {
        mes "[Overlook Fisherman]";
        mes "You can accept and turn in hunting bounties for all 4 dungeon floors directly with me!";
        mes "Completing each floor's quest grants large Base and Job EXP rewards. Finish all 4 floors to claim your choice of a Water Dungeon Card reward!";
        close;
    }

    // BOUNTY QUEST MANAGER
    if (.@menu == 2) {
        mes "[Overlook Fisherman]";
        mes "Which floor bounty would you like to view, accept, or turn in?";
        next;
        .@flr_choice = select("Floor 1 Quest (Lv 30-45):Floor 2 Quest (Lv 45-60):Floor 3 Quest (Lv 60-75):Floor 4 Quest (Lv 75-85):Cancel");

        if (.@flr_choice == 5) close;

        // Prerequisite checks
        if (.@flr_choice == 2 && OWD_Floor1_Done < 1) { mes "[Overlook Fisherman]"; mes "You must clear the Floor 1 quest first!"; close; }
        if (.@flr_choice == 3 && OWD_Floor2_Done < 1) { mes "[Overlook Fisherman]"; mes "You must clear the Floor 2 quest first!"; close; }
        if (.@flr_choice == 4 && OWD_Floor3_Done < 1) { mes "[Overlook Fisherman]"; mes "You must clear the Floor 3 quest first!"; close; }

        if (.@flr_choice == 1) callsub OnHandleOWDQuest, 1, 1068, 1070, 50, 50, 250000, 150000, "Aged Hydra", "Aged Kukre";
        if (.@flr_choice == 2) callsub OnHandleOWDQuest, 2, 1069, 1141, 75, 75, 600000, 400000, "Aged Vadon", "Aged Marina";
        if (.@flr_choice == 3) callsub OnHandleOWDQuest, 3, 1067, 1044, 100, 100, 1200000, 800000, "Aged Obeaune", "Aged Marc";
        if (.@flr_choice == 4) callsub OnHandleOWDQuest, 4, 1269, 1206, 100, 100, 2500000, 1500000, "Aged Strouf", "Aged Merman";
        end;
    }

    // CLAIM FINAL CARD REWARD
    if (.@menu == 3) {
        if (OWD_ClearedAll == 1) {
            mes "[Overlook Fisherman]";
            mes "You have already claimed your grand card reward!";
            close;
        }

        if (OWD_Floor1_Done < 1 || OWD_Floor2_Done < 1 || OWD_Floor3_Done < 1 || OWD_Floor4_Done < 1) {
            mes "[Overlook Fisherman]";
            mes "You must complete all four floor quests first!";
            mes "Floor 1: " + (OWD_Floor1_Done ? "^00FF00Cleared^000000" : "^FF0000Pending^000000");
            mes "Floor 2: " + (OWD_Floor2_Done ? "^00FF00Cleared^000000" : "^FF0000Pending^000000");
            mes "Floor 3: " + (OWD_Floor3_Done ? "^00FF00Cleared^000000" : "^FF0000Pending^000000");
            mes "Floor 4: " + (OWD_Floor4_Done ? "^00FF00Cleared^000000" : "^FF0000Pending^000000");
            close;
        }

        mes "[Overlook Fisherman]";
        mes "Congratulations on clearing all four levels of the Overlook Water Dungeon!";
        mes "As a reward, you may select one card of your choice:";
        next;

        .@card_choice = select("Hydra Card:Vadon Card:Marina Card:Obeaune Card:Marc Card:Merman Card");

        switch(.@card_choice) {
            case 1: .@card_id = 4021; break; // Hydra Card
            case 2: .@card_id = 4070; break; // Vadon Card
            case 3: .@card_id = 4060; break; // Marina Card
            case 4: .@card_id = 4087; break; // Obeaune Card
            case 5: .@card_id = 4105; break; // Marc Card
            case 6: .@card_id = 4130; break; // Merman Card
        }

        OWD_ClearedAll = 1;
        getitem .@card_id, 1;
        specialeffect2 154;

        mes "[Overlook Fisherman]";
        mes "Here is your reward! Use it well in your future adventures.";
        close;
    }

    // WARP TO DUNGEON
    if (OWD_ActiveQuest == 0 && (.@base_lv < 30 || .@base_lv > 85)) {
        mes "[Overlook Fisherman]";
        mes "Your Base Level (" + .@base_lv + ") must be between ^0055FF30 and 85^000000 to enter.";
        close;
    }

    mes "[Overlook Fisherman]";
    mes "Which floor would you like to enter?";
    next;
    .@floor = select("Floor 1 (Lv 30-45):Floor 2 (Lv 45-60):Floor 3 (Lv 60-75):Floor 4 (Lv 75-85):Cancel");

    switch(.@floor) {
        case 1: warp "iz_dun00", 168, 168; end;
        case 2: warp "iz_dun01", 145, 145; end;
        case 3: warp "iz_dun02", 230, 200; end;
        case 4: warp "iz_dun03", 30, 65; end;
        case 5: close;
    }

// Dynamic Quest Handler Subroutine
OnHandleOWDQuest:
    .@q_id = getarg(0);
    .@m1 = getarg(1); .@m2 = getarg(2);
    .@req1 = getarg(3); .@req2 = getarg(4);
    .@b_exp = getarg(5); .@j_exp = getarg(6);
    .@n1$ = getarg(7); .@n2$ = getarg(8);

    if (OWD_ActiveQuest != 0 && OWD_ActiveQuest != .@q_id) {
        mes "[Overlook Fisherman]";
        mes "You already have an active bounty for ^0055FFFloor " + OWD_ActiveQuest + "^000000!";
        mes "Complete or turn in your active bounty before accepting a new one.";
        close;
    }

    if ((.@q_id == 1 && OWD_Floor1_Done) || (.@q_id == 2 && OWD_Floor2_Done) ||
        (.@q_id == 3 && OWD_Floor3_Done) || (.@q_id == 4 && OWD_Floor4_Done)) {
        mes "[Overlook Fisherman]";
        mes "You have already completed the Floor " + .@q_id + " bounty!";
        close;
    }

    // Turn-in & EXP Reward
    if (OWD_Kill1 >= .@req1 && OWD_Kill2 >= .@req2) {
        OWD_Kill1 = 0; OWD_Kill2 = 0; OWD_ActiveQuest = 0;

        if (.@q_id == 1) OWD_Floor1_Done = 1;
        if (.@q_id == 2) OWD_Floor2_Done = 1;
        if (.@q_id == 3) OWD_Floor3_Done = 1;
        if (.@q_id == 4) OWD_Floor4_Done = 1;

        getexp .@b_exp, .@j_exp;
        specialeffect2 154;
        mes "[Overlook Fisherman]";
        mes "^0055FFFloor " + .@q_id + " Training Complete!^000000";
        mes "Granted Rewards:";
        mes "^0055FF+" + .@b_exp + " Base EXP^000000";
        mes "^0055FF+" + .@j_exp + " Job EXP^000000";

        if (OWD_Floor1_Done && OWD_Floor2_Done && OWD_Floor3_Done && OWD_Floor4_Done) {
            next;
            mes "[Overlook Fisherman]";
            mes "^0055FFYou have completed all 4 floors!^000000";
            mes "Speak to me again and choose 'Claim Final Card Reward' to get your card choice!";
        }
        close;
    }

    // Check Active Progress
    if (OWD_ActiveQuest == .@q_id) {
        mes "[Overlook Fisherman]";
        mes "Floor " + .@q_id + " Bounty Progress:";
        mes "- " + .@n1$ + ": ^0055FF" + OWD_Kill1 + "^000000 / " + .@req1;
        mes "- " + .@n2$ + ": ^0055FF" + OWD_Kill2 + "^000000 / " + .@req2;
        close;
    }

    // Accept Quest
    OWD_ActiveQuest = .@q_id;
    OWD_Kill1 = 0; OWD_Kill2 = 0;
    mes "[Overlook Fisherman]";
    mes "Floor " + .@q_id + " Bounty Accepted!";
    mes "Defeat " + .@req1 + "x ^0055FF" + .@n1$ + "^000000 and " + .@req2 + "x ^0055FF" + .@n2$ + "^000000 inside the dungeon.";
    close;
}

// --- Overlook Dungeon Kill Tracker ---
// Spawns live in overlook_mobs.txt (plain map spawn lines, governed
// by dynamic_mobs); each line points its kill event here, so keep
// this NPC name as OWD_MobManager.
-	script	OWD_MobManager	-1,{
OnOWDKill:
    if (OWD_ActiveQuest == 0) end;

    .@k = killedrid;

    if (OWD_ActiveQuest == 1) {
        if (.@k == 1068 && OWD_Kill1 < 50) {
            OWD_Kill1 += 1;
            dispbottom "[Overlook Bounty] Aged Hydra: [" + OWD_Kill1 + "/50]";
        }
        if (.@k == 1070 && OWD_Kill2 < 50) {
            OWD_Kill2 += 1;
            dispbottom "[Overlook Bounty] Aged Kukre: [" + OWD_Kill2 + "/50]";
        }
    }
    else if (OWD_ActiveQuest == 2) {
        if (.@k == 1069 && OWD_Kill1 < 75) {
            OWD_Kill1 += 1;
            dispbottom "[Overlook Bounty] Aged Vadon: [" + OWD_Kill1 + "/75]";
        }
        if (.@k == 1141 && OWD_Kill2 < 75) {
            OWD_Kill2 += 1;
            dispbottom "[Overlook Bounty] Aged Marina: [" + OWD_Kill2 + "/75]";
        }
    }
    else if (OWD_ActiveQuest == 3) {
        if (.@k == 1067 && OWD_Kill1 < 100) {
            OWD_Kill1 += 1;
            dispbottom "[Overlook Bounty] Aged Obeaune: [" + OWD_Kill1 + "/100]";
        }
        if (.@k == 1044 && OWD_Kill2 < 100) {
            OWD_Kill2 += 1;
            dispbottom "[Overlook Bounty] Aged Marc: [" + OWD_Kill2 + "/100]";
        }
    }
    else if (OWD_ActiveQuest == 4) {
        if (.@k == 1269 && OWD_Kill1 < 100) {
            OWD_Kill1 += 1;
            dispbottom "[Overlook Bounty] Aged Strouf: [" + OWD_Kill1 + "/100]";
        }
        if (.@k == 1206 && OWD_Kill2 < 100) {
            OWD_Kill2 += 1;
            dispbottom "[Overlook Bounty] Aged Merman: [" + OWD_Kill2 + "/100]";
        }
    }
    end;
}
