//= Bounty Hunter -- Pre-renewal contracts
//===== By: MondoTruth, Claude =================================
//===== Description: =========================================
//= Replaces npc/when/enable_bounties/bounty_contracts_era.txt on
//= Pre-renewal (app 1.4.3+). The Illusion Dungeons and their
//= monsters do not exist on Pre-renewal (max Base Level 99), so
//= type 3 offers three late Pre-renewal dungeons instead.
//= Monsters are spawned by bounty_mobs_era.txt next to this file.
//===== How to customize: ====================================
//= - Format as in bounty_contracts.txt:
//=   "type|min lv|max lv|map|warp x,y|monster IDs|kills|base|job|zeny"
//= - Warp points are the stock entrances (npc/warps in rAthena).
//= - Rewards are a first guess -- tune after testing.
//============================================================

function	script	F_Bounty_ContractEra	{
	// ---- 3. Hard Dungeons (Pre-renewal stand-in for Illusion) -----------
	setarray .@t$[getarraysize(.@t$)], "3|85|999|abyss_03|115,30|1713,1714|100|400|280|0";
	setarray .@t$[getarraysize(.@t$)], "3|90|999|tha_t10|155,100|1702,1703,1700|100|900|600|0";
	setarray .@t$[getarraysize(.@t$)], "3|95|999|odin_tem03|120,54|1752,1753,1754|100|1400|1000|0";

	// Line N of this table, or "" past the end.
	.@i = getarg(0);
	if (.@i < 0 || .@i >= getarraysize(.@t$)) return "";
	return .@t$[.@i];
}

// Menu name per type ("" = keep the default).
function	script	F_Bounty_EraTypeName	{
	if (getarg(0) == 3) return "Hard Dungeons (Lv 85+)";
	return "";
}
